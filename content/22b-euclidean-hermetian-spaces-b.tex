\section{The Orthogonal Complement}

Let $(V, \langle \cdot, \cdot \rangle)$ be an inner product space over $K$ (where $K = \mathbb{R}$ or $\mathbb{C}$).

% Def. (22.b.1)
\begin{definition}[The Orthogonal Complement]
\label{def:orthogonal_complement}
Let $\emptyset \neq S \subseteq V$ be a subset. We define the \newterm{orthogonal complement} of $S$ as:
\[
    S^\perp := \{v \in V \mid v \perp s \;\forall s \in S\} = \{v \in V \mid \langle v, s \rangle = 0 \;\forall s \in S\}.
\]
When $S = \{v\}$, we write $v^\perp := \{v\}^\perp$.
\end{definition}

% Lemma. (22.b.2)
\begin{lemma}[Properties of the Orthogonal Complement]
\label{lem:orthogonal_complement_properties}
Let $\emptyset \neq S, T \subseteq V$. Then:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $S^\perp \subseteq V$ is a linear subspace.
    \item $0^\perp = V$, and $V^\perp = \{0\}$.
    \item $S \cap S^\perp$ is either $\emptyset$ or equal to $\{0\}$.
    \item If $S \subseteq T$, then $S^\perp \supseteq T^\perp$.
    \item $(\Sp(S))^\perp = S^\perp$.
    \item $S \subseteq (S^\perp)^\perp$.
\end{enumerate}
\end{lemma}

\begin{proof}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item We have $0 \in S^\perp$ because $\langle 0, s \rangle = 0$ for all $s \in S$.
    If $\alpha, \beta \in K$ and $v, w \in S^\perp$, then for all $s \in S$ we have:
    \[
        \langle \alpha v + \beta w, s \rangle = \alpha \langle v, s \rangle + \beta \langle w, s \rangle = 0.
    \]
    This implies that $\alpha v + \beta w \in S^\perp$. Thus, $S^\perp$ is a linear subspace.
    
    \item Showing $0^\perp = V$ is left as an exercise. Let $v \in V^\perp$. Then $\langle v, w \rangle = 0$ for all $w \in V$. In particular, $\langle v, v \rangle = 0$, which implies $v = 0$. This shows $V^\perp \subseteq \{0\}$. Since it is clear that $\{0\} \subseteq V^\perp$, we conclude $V^\perp = \{0\}$.
    
    \item If $S \cap S^\perp = \emptyset$, we are done. Otherwise, assume that $s \in S \cap S^\perp$. This implies that $\langle s, s \rangle = 0$, which yields $s = 0$.
    
    \item (Exercise).
    
    \item We have $S \subseteq \Sp(S)$. By statement \textbf{(d)}, this yields $(\Sp(S))^\perp \subseteq S^\perp$.
    Next, we will show that $S^\perp \subseteq (\Sp(S))^\perp$. Indeed, let $v \in S^\perp$, and let $u \in \Sp(S)$. We can write $u = \sum_{i=1}^k \alpha_i u_i$ with $\alpha_i \in K$ and $u_i \in S$ for all $1 \leq i \leq k$. We compute:
    \[
        \langle v, u \rangle = \sum_{i=1}^k \overline{\alpha_i} \underbrace{\langle v, u_i \rangle}_{= 0} = 0,
    \]
    where we used the fact that $\langle v, u_i \rangle = 0$ because $v \in S^\perp$ and $u_i \in S$. Thus, $v \perp u$. This holds for all $u \in \Sp(S)$, and therefore $v \in (\Sp(S))^\perp$. This shows that $S^\perp \subseteq (\Sp(S))^\perp$.
    
    \item Let $s \in S$. Then for all $v \in S^\perp$ we have $\langle v, s \rangle = 0$, which implies $\langle s, v \rangle = 0$. Therefore, $s \perp v$ for all $v \in S^\perp$. Consequently, $s \in (S^\perp)^\perp$.
\end{enumerate}
\end{proof}

% Thm. (22.b.3)
\begin{theorem}[Orthogonal Complement in Finite Dimensions]
\label{thm:orthogonal_complement_finite_dim}
Let $V$ be an inner product space (possibly of infinite dimension), and let $U \subseteq V$ be a finite-di\-men\-sional subspace. Then $U^\perp$ is a complement of $U$ in $V$ (i.e., $U + U^\perp = V$ and $U \cap U^\perp = \{0\}$).
\end{theorem}

\begin{proof}
By \cref{lem:orthogonal_complement_properties} \textbf{(c)}, the intersection $U \cap U^\perp$ is either empty or $\{0\}$; it is not empty, since $U$ is a subspace and so contains $0$, which lies in $U^\perp$ as well. Hence, $U \cap U^\perp = \{0\}$, and it is enough to show that $U + U^\perp = V$.
Put $r := \dim U$ and pick an orthonormal basis $(e_1, \dots, e_r)$ for $U$, which exists by \cref{cor:orthonormal_basis_exists} because $U$ is finite-di\-men\-sional. (Note: this is not necessarily the standard basis or anything similar). Let $v \in V$.
We introduce the temporary notation $\widetilde{P}_U(v) := \langle v, e_1 \rangle e_1 + \dots + \langle v, e_r \rangle e_r \in U$. (This is the orthogonal projection of $v$ onto $U$.)
We have
\begin{equation}
\label{eq:v_decomposition_temp}
    v = \underbrace{\widetilde{P}_U(v)}_{\in U} + (v - \widetilde{P}_U(v)).
\end{equation}
We claim that $v - \widetilde{P}_U(v) \in U^\perp$. Indeed, for any $1 \leq i \leq r$,
\begin{align*}
    \langle v - \widetilde{P}_U(v), e_i \rangle &= \langle v, e_i \rangle - \sum_{k=1}^r \langle v, e_k \rangle \langle e_k, e_i \rangle \\
    &= \langle v, e_i \rangle - \langle v, e_i \rangle = 0.
\end{align*}
This holds for all $1 \leq i \leq r$, hence $(v - \widetilde{P}_U(v)) \perp \Sp(e_1, \dots, e_r) = U$, meaning $v - \widetilde{P}_U(v) \in U^\perp$.
Going back to \eqref{eq:v_decomposition_temp}, we see that
\[
    v = \underbrace{\widetilde{P}_U(v)}_{\in U} + \underbrace{(v - \widetilde{P}_U(v))}_{\in U^\perp}.
\]
This holds for all $v \in V$, which implies $V = U + U^\perp$.
\end{proof}

\begin{remark}
If $U \subseteq V$ is a finite-di\-men\-sional subspace, then since $U^\perp$ is a complement of $U$ in $V$, we have a canonical isomorphism $V \cong U \oplus U^\perp$.
\end{remark}

% Def. (22.b.4)
\begin{definition}[Orthogonal Projection]
\label{def:orthogonal_projection}
Let $V$ be an inner product space and $U \subseteq V$ a finite-di\-men\-sional subspace. We have seen that $U^\perp$ is a complement of $U$ in $V$ (which implies that for all $v \in V$, there exist unique vectors $u \in U$ and $w \in U^\perp$ such that $v = u + w$).
Define a map $P_U \colon V \to U$ as follows: for all $v \in V$, write $v = u + w$ with $u \in U$ and $w \in U^\perp$ in a unique way. We define $P_U(v) := u$. The map $P_U$ is called the \newterm{orthogonal projection} on $U$.
\end{definition}

% Prop. (22.b.5)
\begin{proposition}[Properties of Orthogonal Projection]
\label{prop:orthogonal_projection_properties}
The map $P_U$ has the following properties:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $P_U$ is a linear map.
    \item $\Img P_U = U$ and $\ker P_U = U^\perp$.
    \item For all $v \in V$, $v - P_U(v) \in U^\perp$.
    \item For all $u \in U$, $P_U(u) = u$.
    \item If $(e_1, \dots, e_r)$ is any orthonormal basis for $U$, then $P_U(v) = \sum_{i=1}^r \langle v, e_i \rangle \cdot e_i$ (which exactly equals $\widetilde{P}_U(v)$ from the previous proof).
\end{enumerate}
\end{proposition}

\begin{proof}
Properties \textbf{(a)}--\textbf{(d)} follow from the general fact that if $W_1, W_2$ are vector subspaces of $W$ and $W_1 \oplus W_2 = W$, then the map $P_{W_1} \colon W \to W_1$, defined by $P_{W_1}(w) = w_1$, where we write $w = w_1 + w_2$ in a unique way with $w_1 \in W_1, w_2 \in W_2$, is a linear map. (That map is well defined precisely because the decomposition is unique, which is \cref{cor:trivial_intersection} \textbf{(d)}. It is linear because if $w = w_1 + w_2$ and $w' = w_1' + w_2'$ are the decompositions of $w$ and $w'$, then $\alpha w + \beta w' = (\alpha w_1 + \beta w_1') + (\alpha w_2 + \beta w_2')$ exhibits a decomposition of $\alpha w + \beta w'$ with the first summand in $W_1$ and the second in $W_2$; by uniqueness it \emph{is} the decomposition, so $P_{W_1}(\alpha w + \beta w') = \alpha w_1 + \beta w_1' = \alpha P_{W_1}(w) + \beta P_{W_1}(w')$.) Furthermore, $\Img P_{W_1} = W_1$, $\ker P_{W_1} = W_2$, and for all $w \in W$ we have $w - P_{W_1}(w) \in W_2$. (Thus, $w = P_{W_1}(w) + (w - P_{W_1}(w))$ is the unique decomposition of $w$ as $w = w_1 + w_2$). We also have $P_{W_1}(w) = w$ for all $w \in W_1$.

For statement \textbf{(e)}, we can write
\begin{equation}
\label{eq:orthogonal_decomposition_v}
    v = \underbrace{\left( \sum_{i=1}^r \langle v, e_i \rangle \cdot e_i \right)}_{\text{(I)}} + \underbrace{\left( v - \sum_{i=1}^r \langle v, e_i \rangle e_i \right)}_{\text{(II)}}.
\end{equation}
Now, part \textbf{(I)} is exactly $\widetilde{P}_U(v)$ from the previous proof, and we have also seen in that proof that part \textbf{(II)} belongs to $U^\perp$. This implies that \eqref{eq:orthogonal_decomposition_v} is the unique decomposition of $v$ with respect to $V \cong U \oplus U^\perp$. Hence, $P_U(v) = \text{(I)}$.
\end{proof}

\begin{exercise}[Best Approximation Theorem \faHeart]
  \label{exc:best_approximation_subspace}
  Let $V$ be an inner product space, let $U \subseteq V$ be a finite-dimensional subspace, and let $P_U \colon V \to U$ denote the orthogonal projection onto $U$. Let $v \in V$. Show that for every $u \in U$,
  \[
  \|v - P_U(v)\| \le \|v - u\|,
  \]
  with equality if and only if $u = P_U(v)$. Thus, $P_U(v)$ is the unique point in $U$ closest to $v$.
  \exhint[Hint (Y. St\"aubli)]{Write $v - u = (v - P_U(v)) + (P_U(v) - u)$. Note that $P_U(v) - u \in U$, while $v - P_U(v) \in U^\perp$ by \cref{prop:orthogonal_projection_properties}(c). % Cor. (22.b.6)
\begin{corollary}[Bidual Orthogonal Complement]
\label{cor:bidual_orthogonal_complement}
Let $V$ be an inner product space and let $U \subseteq V$ be a finite-di\-men\-sional subspace. Then $(U^\perp)^\perp = U$.
\end{corollary}

\begin{proof}
By \cref{lem:orthogonal_complement_properties}, we have $U \subseteq (U^\perp)^\perp$. We will show that $(U^\perp)^\perp \subseteq U$ also holds. Indeed, let $v \in (U^\perp)^\perp$.
Recall that $U^\perp$ is a complement of $U$ in $V$.
Write $v = u + w$ with $u \in U$ and $w \in U^\perp$.
Since $u \in U \subseteq (U^\perp)^\perp$ and also $v \in (U^\perp)^\perp$, we have that $w = v - u \in (U^\perp)^\perp$. But $w \in U^\perp$, hence $w \in U^\perp \cap (U^\perp)^\perp = \{0\}$.
This shows that $w = 0$, and therefore, $v = u \in U$. We have proven that $(U^\perp)^\perp \subseteq U$.
\end{proof}

% Cor. (22.b.7)
\begin{corollary}[Dimension Formula for the Orthogonal Complement]
\label{cor:dim_orthogonal_complement}
If $V$ is a finite-di\-men\-sional inner product space and $U \subseteq V$ is a subspace, then $\dim U + \dim U^\perp = \dim V$.
\end{corollary}

% Creator: Opus 5
\begin{proof}
Since $V$ is finite-di\-men\-sional, so is its subspace $U$, and \cref{thm:orthogonal_complement_finite_dim} applies: $U + U^\perp = V$ and $U \cap U^\perp = \{0\}$, so the sum $U + U^\perp$ is direct and equals $V$. Choosing a basis for $U$ and a basis for $U^\perp$, both of which exist because $U$ and $U^\perp$ are subspaces of the finite-di\-men\-sional $V$ (\cref{prop:subspace_dimension_constraints}), and concatenating them therefore produces a basis for $V$, by the basis criterion of \cref{ex:direct_sum_conditions}, whence
\[
    \dim U + \dim U^\perp = \dim \left( U \oplus U^\perp \right) = \dim V. \qedhere
\]
\end{proof}

\subsection*{Orthogonal and Unitary Matrices}

% Def. (22.b.8)
\begin{definition}[Orthogonal and Unitary Matrices]
\label{def:orthogonal_unitary_matrices}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item A matrix $A \in \M_{n \times n}(\mathbb{R})$ is called \newterm{orthogonal} if its columns form an orthonormal system in $\mathbb{R}^n$, with respect to the standard scalar product of $\mathbb{R}^n$. That is, writing $A$ in terms of its columns,
    \[
        A = \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix},
    \]
    the condition means that $\langle v_i, v_j \rangle = \delta_{ij}$ for all $1 \leq i, j \leq n$ (which in particular implies that the columns form an orthonormal basis for $\mathbb{R}^n$). We denote the set of orthogonal matrices of size $n \times n$ by $\Orth(n) \subseteq \M_{n \times n}(\mathbb{R})$.
    \item A matrix $A \in \M_{n \times n}(\mathbb{C})$ is called \newterm{unitary} if its columns form an orthonormal system in $\mathbb{C}^n$, with respect to the standard Hermitian product on $\mathbb{C}^n$. We denote the set of all unitary matrices of size $n \times n$ by $\Unit(n) \subseteq \M_{n \times n}(\mathbb{C})$.
\end{enumerate}
\end{definition}

\begin{remark}
Why are they called \qt{orthogonal matrices} and not \qt{orthonormal matrices}? This is simply a historical convention in the terminology!
\end{remark}

% Lemma. (22.b.9)
\begin{lemma}[Characterization of Orthogonal Matrices]
\label{lem:char_orthogonal_matrices}
For a matrix $A \in \M_{n \times n}(\mathbb{R})$, the following statements are equivalent:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $A$ is an orthogonal matrix.
    \item $A\transp A = I_n$.
    \item $A A\transp = I_n$.
    \item $A^{-1} = A\transp$.
\end{enumerate}
\end{lemma}

% Lemma. (22.b.10)
\begin{lemma}[Characterization of Unitary Matrices]
\label{lem:char_unitary_matrices}
For a matrix $A \in \M_{n \times n}(\mathbb{C})$, the following statements are equivalent:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $A$ is a unitary matrix.
    \item $A^* A = I_n$ (which is equivalent to $\overline{A}\transp A = I_n$).
    \item $A A^* = I_n$ (which is equivalent to $A \overline{A}\transp = I_n$).
    \item $A^{-1} = A^*$.
\end{enumerate}
\end{lemma}

\begin{proof}[Proof of \cref{lem:char_unitary_matrices}]
(The proof of \cref{lem:char_orthogonal_matrices} is extremely similar).
Write
\[
    A = \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix},
\]
where $v_k \in \mathbb{C}_{\text{col}}^n$ for $1 \leq k \leq n$. We compute the product:
% \begin{equation}
% \label{eq:unitary_matrix_product}
%     A^* A = \begin{pmatrix} \text{---} & \overline{v_1}\transp & \text{---} \\ & \vdots & \\ \text{---} & \overline{v_n}\transp & \text{---} \end{pmatrix} \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix} = \left( \overline{v_i}\transp \cdot v_j \right)_{1 \leq i \leq n, 1 \leq j \leq n} = \left( \langle v_j, v_i \rangle \right)_{1 \leq i \leq n, 1 \leq j \leq n}.
% \end{equation}
\begin{align}
    A^* A &= \begin{pmatrix} \text{---} & \overline{v_1}\transp & \text{---} \\ & \vdots & \\ \text{---} & \overline{v_n}\transp & \text{---} \end{pmatrix} \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix} \nonumber \\
    &= \left( \overline{v_i}\transp \cdot v_j \right)_{1 \leq i, j \leq n} \nonumber \\
    \label{eq:unitary_matrix_product}
    &= \left( \langle v_j, v_i \rangle \right)_{1 \leq i, j \leq n}.
\end{align}
Now, $A$ is unitary if and only if $v_1, \dots, v_n$ form an orthonormal system, which is equivalent to saying that the resulting matrix in \eqref{eq:unitary_matrix_product} is exactly $I_n$. This proves \textbf{(a)} $\iff$ \textbf{(b)}.

% Extractor: Opus 5
For the remaining equivalences, suppose $A^* A = I_n$. Taking determinants and using $\det(A^*) = \det(\overline{A}\transp) = \det(\overline{A}) = \overline{\det(A)}$, we obtain $\left| \det(A) \right|^2 = 1$, so $\det(A) \ne 0$ and $A$ is invertible. Multiplying $A^* A = I_n$ on the right by $A^{-1}$ gives $A^* = A^{-1}$, which is \textbf{(d)}; and then $A A^* = A A^{-1} = I_n$, which is \textbf{(c)}. Conversely, \textbf{(c)} gives by the same argument $A^* = A^{-1}$ and hence \textbf{(b)}, so all four statements are equivalent.
% Extractor: Gemini 3.1 Pro
\end{proof}

\begin{corollary}[Rows of Orthogonal and Unitary Matrices]
\label{cor:rows_orthogonal_unitary}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $A \in \M_{n \times n}(\mathbb{R})$. The matrix $A \in \Orth(n)$ if and only if $A\transp \in \Orth(n)$, which holds if and only if the rows of $A$ form an orthonormal basis for $\mathbb{R}^n$.
    \item Let $A \in \M_{n \times n}(\mathbb{C})$. The matrix $A \in \Unit(n)$ if and only if $A^* \in \Unit(n)$, which holds if and only if the rows of $A$ form an orthonormal basis for $\mathbb{C}^n$.
\end{enumerate}
\end{corollary}

% Creator: Opus 5
\begin{proof}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item By \cref{lem:char_orthogonal_matrices}, $A \in \Orth(n)$ if and only if $A\transp A = I_n$, and equally if and only if $A A\transp = I_n$. Applying the same characterization to the matrix $A\transp$, whose transpose is $A$, the condition $A\transp \in \Orth(n)$ reads $(A\transp)\transp A\transp = A A\transp = I_n$, the very same equation. Hence, $A \in \Orth(n) \iff A\transp \in \Orth(n)$. Finally, the columns of $A\transp$ are exactly the rows of $A$, so $A\transp \in \Orth(n)$ says precisely that the rows of $A$ form an orthonormal system, and an orthonormal system of $n$ vectors in $\mathbb{R}^n$ is a basis by \cref{thm:properties_orthogonal_systems} \textbf{(a)}.
    \item Identical, with \cref{lem:char_unitary_matrices} in place of \cref{lem:char_orthogonal_matrices} and $A^*$ in place of $A\transp$: the condition $A^* \in \Unit(n)$ reads $(A^*)^* A^* = A A^* = I_n$, which is one of the equivalent forms of $A \in \Unit(n)$; and the columns of $A^*$ are the conjugates of the rows of $A$, which form an orthonormal system exactly when the rows do. \qedhere
\end{enumerate}
\end{proof}

\begin{theorem}[QR-Decomposition for Invertible Matrices]
\label{thm:qr_decomposition_invertible}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $A \in \GL_n(\mathbb{R})$. Then there exists $Q \in \Orth(n)$ and an upper triangular matrix $R \in \M_{n \times n}(\mathbb{R})$ such that $A = Q \cdot R$.
    \item Let $A \in \GL_n(\mathbb{C})$. Then there exists $Q \in \Unit(n)$ and an upper triangular matrix $R \in \M_{n \times n}(\mathbb{C})$ such that $A = Q \cdot R$.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of \cref{thm:qr_decomposition_invertible}]
We prove parts \textbf{(a)} and \textbf{(b)} together, writing $K = \mathbb{R}$ or $K = \mathbb{C}$ throughout; the only difference between the two cases is whether the resulting $Q$ is called orthogonal or unitary.

Let $A \in \GL_n(K)$ and write
\[
    A := \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix}
\]
with $v_k \in K_{\text{col}}^n$. Since $A \in \GL_n(K)$, the vectors $v_1, \dots, v_n$ form a basis for $K^n$. Apply the Gram-Schmidt orthogonalization process (\cref{thm:gram_schmidt}) to $v_1, \dots, v_n$ to obtain an orthonormal basis $e_1, \dots, e_n$ for $K^n$, and recall from part \textbf{(c)} of that theorem that
\begin{equation} \label{eq:qr_nested_spans}
    \Sp(v_1, \dots, v_j) = \Sp(e_1, \dots, e_j) \qquad \forall \; 1 \le j \le n.
\end{equation}

% Extractor: Opus 5
Fix $1 \le j \le n$. By \eqref{eq:qr_nested_spans} the vector $v_j$ lies in $\Sp(e_1, \dots, e_j)$, and since $(e_1, \dots, e_j)$ is orthonormal, \cref{thm:properties_orthogonal_systems} \textbf{(b)} identifies its coefficients: they are $\langle v_j, e_i \rangle / \langle e_i, e_i \rangle = \langle v_j, e_i \rangle$. So, the whole list expands as
\begin{align*}
    v_1 &= \langle v_1, e_1 \rangle e_1, \\
    v_2 &= \langle v_2, e_1 \rangle e_1 + \langle v_2, e_2 \rangle e_2, \\
        &\;\;\vdots \\
    v_j &= \langle v_j, e_1 \rangle e_1 + \dots + \langle v_j, e_j \rangle e_j, \\
        &\;\;\vdots \\
    v_n &= \langle v_n, e_1 \rangle e_1 + \dots + \langle v_n, e_n \rangle e_n,
\end{align*}
with no $e_i$ of index $i > j$ ever appearing in $v_j$. Read column by column, these $n$ identities are exactly one matrix product:
\[
    \underbrace{\begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix}}_{A}
    = \underbrace{\begin{pmatrix} | & & | \\ e_1 & \dots & e_n \\ | & & | \end{pmatrix}}_{Q}
      \underbrace{\begin{pmatrix}
        \langle v_1, e_1 \rangle & \langle v_2, e_1 \rangle & \dots & \langle v_n, e_1 \rangle \\
        0 & \langle v_2, e_2 \rangle & \dots & \langle v_n, e_2 \rangle \\
        \vdots & & \ddots & \vdots \\
        0 & 0 & \dots & \langle v_n, e_n \rangle
      \end{pmatrix}}_{R},
\]
that is, $R_{ij} = \langle v_j, e_i \rangle$ for $i \le j$ and $R_{ij} = 0$ for $i > j$.

The columns of $Q$ are $e_1, \dots, e_n$, an orthonormal basis of $K^n$, so $Q \in \Orth(n)$ when $K = \mathbb{R}$ and $Q \in \Unit(n)$ when $K = \mathbb{C}$ (\cref{def:orthogonal_unitary_matrices}); and $R$ is upper triangular by construction. Hence, $A = Q \cdot R$ as required.
% Extractor: Gemini 3.7 Flash (High)
\end{proof}

\begin{theorem*}[Extension of QR-Decomposition for All Matrices]
\label{thm:qr_decomposition_all}
Let $A \in \M_{n \times n}(K)$, and let $r := \rank(A)$. Then there exist matrices $Q$ and $R$ with $Q \in \Orth(n)$ if $K = \mathbb{R}$ (and $Q \in \Unit(n)$ if $K = \mathbb{C}$), and $R = \left(\begin{array}{c|c} C & * \\ \hline 0 & 0 \end{array}\right) \in \M_{n \times n}(K)$, where $C \in \M_{r \times r}(K)$ is an upper triangular matrix, such that $A = Q \cdot R$.
\end{theorem*}

\begin{proof}
Let
\[
    A := \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix}.
\]
Note that $r = \dim \Sp(v_1, \dots, v_n)$.
We may assume $r \geq 1$, because otherwise $r = 0$, meaning $A = 0$, in which case we can take $R = 0$ and $Q$ to be any arbitrary orthogonal/unitary matrix.
Note that there exist indices $1 \leq i_1 < \dots < i_r \leq n$ such that:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $v_{i_1}, \dots, v_{i_r}$ are linearly independent.
    \item $v_{i_1}, \dots, v_{i_k}$ is a basis for $\Sp(v_1, v_2, \dots, v_{i_k})$ (and $k = \dim \Sp(v_1, v_2, \dots, v_{i_k}) \leq i_k$).
\end{enumerate}
To see this, let $1 \leq i_1$ be the minimal index $j$ such that $v_j \neq 0$. (Since $r \geq 1$, there exists such an index.) Clearly, $\Sp(v_1, \dots, v_{i_1}) = \Sp(v_{i_1})$. Suppose we have defined $1 \leq i_1 < \dots < i_k < n$ as above. Consider now the minimal index $\ell$, with $i_k < \ell \leq n$ and such that $v_\ell \notin \Sp(v_{i_1}, \dots, v_{i_k})$. If there is no such $\ell$, then $k = r$ and we are done. Otherwise, take $i_{k+1} := \ell$. Continue the construction by induction.

Now, apply the Gram-Schmidt process to the list of vectors $v_{i_1}, \dots, v_{i_r}$ and obtain an orthonormal system $e_1, \dots, e_r$ such that $(e_1, \dots, e_k)$ is a basis for $\Sp(v_{i_1}, \dots, v_{i_k}) = \Sp(v_1, v_2, \dots, v_{i_k})$.
Extend $e_1, \dots, e_r$ to an orthonormal basis $(e_1, \dots, e_r, e_{r+1}, \dots, e_n)$ of $K^n$ (how? See \cref{exc:extend_orthonormal_system}). Take
\[
    Q := \begin{pmatrix} | & & | \\ e_1 & \dots & e_n \\ | & & | \end{pmatrix},
\]
and let $R$ be built exactly as in the previous proof, by $R_{ij} := \langle v_j, e_i \rangle$.

% Extractor: Opus 5
The block shape of $R$ is now forced. Every $v_j$ lies in $\Sp(v_1, \dots, v_j) = \Sp(e_1, \dots, e_{k(j)})$, where $k(j) \le r$ counts how many of the indices $i_1 < \dots < i_r$ do not exceed $j$; in particular no $e_i$ with $i > r$ ever occurs, so the last $n-r$ rows of $R$ vanish. Among the first $r$ rows, the columns $j = i_1, \dots, i_r$ carry the upper triangular block $C := \left( \langle v_{i_j}, e_i \rangle \right)_{1 \le i, j \le r}$, upper triangular for the same reason as before, while the remaining columns are the arbitrary entries denoted by $*$. Hence, $A = Q \cdot R$ with $R$ of the stated form.
% Extractor: Gemini 3.1 Pro
\end{proof}

\begin{exercisesolution}[Proof of \cref{lem:orthogonal_complement_properties} \textbf{(b)}, the identity $0^\perp = V$]
\label{sol:orthogonal_complement_properties_b}
  Here $0$ denotes the subset $\{0\} \subseteq V$, so by \cref{def:orthogonal_complement}
  \[ 0^\perp = \{ v \in V \mid \langle v, 0 \rangle = 0 \}. \]
  By \cref{lem:basic_properties_inner_prod} \textbf{(a)} we have $\langle v, 0 \rangle = 0$ for \emph{every} $v \in V$, so the defining condition is satisfied by every vector of $V$ and $0^\perp = V$.

  The companion statement $V^\perp = \{0\}$, proved in the text, is the mirror image: the larger the set, the more orthogonality conditions, and the smaller its complement.
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{lem:orthogonal_complement_properties} \textbf{(d)}]
\label{sol:orthogonal_complement_properties_d}
  Assume $S \subseteq T$ and let $v \in T^\perp$, so that $\langle v, t \rangle = 0$ for every $t \in T$. Every $s \in S$ is in particular an element of $T$, so $\langle v, s \rangle = 0$ for every $s \in S$, that is, $v \in S^\perp$. Hence, $T^\perp \subseteq S^\perp$, which is the assertion $S^\perp \supseteq T^\perp$.

  Note that the inclusion reverses: passing to a larger set imposes more conditions on $v$, so fewer vectors survive.
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{exc:char_orthogonal_matrices_proof}, that is, of \cref{lem:char_orthogonal_matrices}]
\label{sol:char_orthogonal_matrices_proof}
  Write
  \[
      A = \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix}, \qquad v_k \in \mathbb{R}^n_{\text{col}},
  \]
  and recall that the standard scalar product on $\mathbb{R}^n$ is $\langle u, v \rangle = u\transp v$.

  \textbf{(a)} $\iff$ \textbf{(b)}. The rows of $A\transp$ are the vectors $v_1\transp, \dots, v_n\transp$, so
  \[
      A\transp A = \begin{pmatrix} \text{---} & v_1\transp & \text{---} \\ & \vdots & \\ \text{---} & v_n\transp & \text{---} \end{pmatrix} \begin{pmatrix} | & & | \\ v_1 & \dots & v_n \\ | & & | \end{pmatrix} = \left( v_i\transp v_j \right)_{1 \le i, j \le n} = \left( \langle v_i, v_j \rangle \right)_{1 \le i, j \le n}.
  \]
  By \cref{def:orthogonal_unitary_matrices}, $A$ is orthogonal exactly when $\langle v_i, v_j \rangle = \delta_{ij}$ for all $i, j$, which says precisely that the matrix just computed is $I_n$.

  This is where the conjugations of the Hermitian case have gone. There the left factor was $\overline{A}\transp$, so the $(i,j)$ entry came out as $\overline{v_i}\transp v_j = \langle v_j, v_i \rangle$, the arguments in the opposite order. Over $\mathbb{R}$ there is no conjugation, the scalar product is symmetric, and the entry is $\langle v_i, v_j \rangle$. Since the condition imposed is the symmetric one $\delta_{ij}$, the difference is invisible in the conclusion, but it is real in the computation.

  \textbf{(b)} $\implies$ \textbf{(d)}. Suppose $A\transp A = I_n$. Taking determinants and using $\det(A\transp) = \det(A)$ gives $\det(A)^2 = 1$, so $\det(A) = \pm 1 \ne 0$ and $A$ is invertible. Multiplying $A\transp A = I_n$ on the right by $A^{-1}$ yields $A\transp = A^{-1}$.

  \textbf{(d)} $\implies$ \textbf{(c)}. If $A\transp = A^{-1}$, then $A A\transp = A A^{-1} = I_n$.

  \textbf{(c)} $\implies$ \textbf{(b)}. Suppose $A A\transp = I_n$. Determinants again give $\det(A)^2 = 1$, so $A$ is invertible; multiplying on the left by $A^{-1}$ gives $A\transp = A^{-1}$, and hence $A\transp A = A^{-1} A = I_n$.

  The four statements are therefore equivalent.
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{exc:extend_orthonormal_system}]
\label{sol:extend_orthonormal_system}
  An orthonormal system is in particular an orthogonal system, hence linearly independent by \cref{thm:properties_orthogonal_systems} \textbf{(a)}. So, $(e_1, \dots, e_r)$ can be extended to a basis $(e_1, \dots, e_r, u_{r+1}, \dots, u_n)$ of $V$.

  Now apply the Gram-Schmidt process (\cref{thm:gram_schmidt}) to that basis, producing $w_1, \dots, w_n$. The point is that Gram-Schmidt does not disturb an initial segment that is already orthonormal. Indeed, $w_1 = e_1$, and if $2 \le j \le r$ and $w_i = e_i$ for all $i < j$, then
  \[ w_j = e_j - \sum_{i=1}^{j-1} \frac{\langle e_j, w_i \rangle}{\langle w_i, w_i \rangle} w_i = e_j - \sum_{i=1}^{j-1} \frac{\langle e_j, e_i \rangle}{1} \, e_i = e_j, \]
  because $\langle e_j, e_i \rangle = 0$ for $i \ne j$ and $\langle e_i, e_i \rangle = 1$. By induction, $w_i = e_i$ for all $1 \le i \le r$.

  Therefore $(w_1, \dots, w_n) = (e_1, \dots, e_r, w_{r+1}, \dots, w_n)$ is an orthogonal basis of $V$, and normalizing only the tail (the first $r$ vectors already have norm $1$) gives the orthonormal basis
  \[ \left( e_1, \dots, e_r, \frac{w_{r+1}}{\|w_{r+1}\|}, \dots, \frac{w_n}{\|w_n\|} \right) \]
  extending the given system.

  There is a second route, worth recording because it is the one the geometry suggests: put $U := \Sp(e_1, \dots, e_r)$. By \cref{cor:dim_orthogonal_complement}, $\dim U^\perp = n - r$; choose an orthonormal basis $(f_{r+1}, \dots, f_n)$ of $U^\perp$, which exists by \cref{cor:orthonormal_basis_exists}. Every $f_i$ is orthogonal to every $e_j$ by the definition of $U^\perp$, so the concatenation $(e_1, \dots, e_r, f_{r+1}, \dots, f_n)$ is an orthonormal system of $n$ vectors, hence linearly independent, hence a basis of $V$.
\end{exercisesolution}

\begin{exercisesolution}[Solution of \cref{exc:gram_schmidt_dependent}]
\label{sol:gram_schmidt_dependent}
  \textbf{The next vector is $0$.} Suppose that after $k$ successful steps we have an orthogonal system $e_1, \dots, e_k$ with
  \[ U := \Sp(e_1, \dots, e_k) = \Sp(v_1, \dots, v_{i_k}), \]
  and that the next vector to be processed satisfies $v_{i_{k+1}} \in \Sp(v_1, \dots, v_{i_k}) = U$. The Gram-Schmidt formula prescribes
  \[ w := v_{i_{k+1}} - \sum_{i=1}^{k} \frac{\langle v_{i_{k+1}}, e_i \rangle}{\langle e_i, e_i \rangle} \, e_i . \]
  But $(e_1, \dots, e_k)$ is an orthogonal system spanning $U$, and $v_{i_{k+1}}$ lies in $U$, so \cref{thm:properties_orthogonal_systems} \textbf{(b)} tells us the coefficients of $v_{i_{k+1}}$ in that system exactly:
  \[ v_{i_{k+1}} = \sum_{i=1}^{k} \frac{\langle v_{i_{k+1}}, e_i \rangle}{\langle e_i, e_i \rangle} \, e_i . \]
  The subtracted sum is therefore $v_{i_{k+1}}$ itself; it is the orthogonal projection $P_U(v_{i_{k+1}})$, which fixes $U$ pointwise by \cref{prop:orthogonal_projection_properties} \textbf{(d)}, and consequently
  \[ w = v_{i_{k+1}} - v_{i_{k+1}} = 0 . \]

  \textbf{What this means for the process.} The construction then stalls: the next step would divide by $\langle w, w \rangle = 0$. Geometrically, this is exactly right, $v_{i_{k+1}}$ contributes no new direction, so there is nothing left after removing its component along $U$.

  \textbf{The remedy.} Discard every vector that produces $0$ and carry on with the next one. The vectors that survive are precisely a maximal linearly independent sublist $v_{i_1}, \dots, v_{i_r}$ of $v_1, \dots, v_m$, where $r = \dim \Sp(v_1, \dots, v_m)$, and the output $e_1, \dots, e_r$ is an orthogonal basis of $\Sp(v_1, \dots, v_m)$ satisfying $\Sp(e_1, \dots, e_k) = \Sp(v_1, \dots, v_{i_k})$ for every $k$.

  This is what the proof of the extension of the QR-decomposition on \cpageref{thm:qr_decomposition_all} arranges in advance: rather than letting the process stall, it selects the indices $i_1 < \dots < i_r$ first and applies Gram-Schmidt only to $v_{i_1}, \dots, v_{i_r}$, which are linearly independent by construction.
\end{exercisesolution}

and $P_U(v) := u$ is well defined for that reason alone. Parts \textbf{(a)} to
\textbf{(d)} of \cref{prop:orthogonal_projection_properties} are then nothing
but the general facts about projecting along a direct sum, and both the
well-definedness and the linearity come from the uniqueness in
\cref{cor:trivial_intersection} \textbf{(d)}. Only part \textbf{(e)}, the
formula $P_U(v) = \sum_{i} \langle v, e_i \rangle e_i$, touches the inner
product at all, which is why it alone asks for an orthonormal basis.

That division is worth carrying into \cref{exc:best_approximation_subspace}.
The geometric content there, that $P_U(v)$ is the point of $U$ nearest to $v$,
comes from part \textbf{(c)} and Pythagoras and never from the formula; the
formula is only how one computes the point once it is known which point it is.

One convention runs through the section. $U$ is assumed finite-di\-men\-sional
while $V$ need not be, and that asymmetry is not idle:
\cref{thm:orthogonal_complement_finite_dim} builds the decomposition out of a
finite orthonormal basis of $U$, so the dimension count of
\cref{cor:dim_orthogonal_complement}, which does need all of $V$ finite, is a
strictly narrower statement than the theorem above it.
\end{ainote}
