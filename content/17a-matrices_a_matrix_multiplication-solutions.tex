\begin{exercisesolution}[Solution to \cref{exc:matrix_powers_2x2}]
\label{sol:matrix_powers_2x2}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $A = \left(\begin{smallmatrix} a & b \\ c & d \end{smallmatrix}\right) \in \M_{2 \times 2}(\mathbb{R})$.
    Computing the matrix square:
    \[
    A^2 = \begin{pmatrix} a & b \\ c & d \end{pmatrix} \begin{pmatrix} a & b \\ c & d \end{pmatrix} = \begin{pmatrix} a^2 + bc & ab + bd \\ ca + dc & cb + d^2 \end{pmatrix} = \begin{pmatrix} a^2 + bc & b(a + d) \\ c(a + d) & bc + d^2 \end{pmatrix}.
    \]

    \item Let $T = \left(\begin{smallmatrix} 1 & 1 \\ 0 & 1 \end{smallmatrix}\right)$.
    We claim that for all $n \in \mathbb{N}$:
    \[
    T^n = \begin{pmatrix} 1 & n \\ 0 & 1 \end{pmatrix}.
    \]
    We prove this formula by mathematical induction on $n \ge 1$:
    \begin{itemize}
        \item \emph{Base case ($n = 1$):} $T^1 = \left(\begin{smallmatrix} 1 & 1 \\ 0 & 1 \end{smallmatrix}\right)$, which matches the formula with $n = 1$.
        \item \emph{Induction step:} Assume $T^n = \left(\begin{smallmatrix} 1 & n \\ 0 & 1 \end{smallmatrix}\right)$ holds for some $n \ge 1$. Then:
        \[
        T^{n+1} = T^n \cdot T = \begin{pmatrix} 1 & n \\ 0 & 1 \end{pmatrix} \begin{pmatrix} 1 & 1 \\ 0 & 1 \end{pmatrix} = \begin{pmatrix} 1\cdot 1 + n \cdot 0 & 1\cdot 1 + n \cdot 1 \\ 0 \cdot 1 + 1 \cdot 0 & 0 \cdot 1 + 1 \cdot 1 \end{pmatrix} = \begin{pmatrix} 1 & n+1 \\ 0 & 1 \end{pmatrix}.
        \]
        This completes the induction step. \qedhere
    \end{itemize}
\end{enumerate}
\exinfo{Official Solution (Problem Sheet 11, Exercise 1). Computes 2x2 matrix square and proves the formula for powers of an upper triangular shear matrix by induction.}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{exc:block_diagonal_upper_triangular_mult}]
\label{sol:block_diagonal_upper_triangular_mult}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Write $A = (a_{ij})$ and $B = (b_{ij})$ in $\M_{n \times n}(K)$, and split the index range $\{1, \dots, n\}$ into the two blocks
    \[
        I := \{1, \dots, k\}, \qquad J := \{k+1, \dots, n\}.
    \]
    The block form of $A$ says precisely that $a_{ij} = 0$ whenever $i$ and $j$ lie in \emph{different} blocks, and likewise for $B$. By \cref{def:matrix_multiplication}, the $(i,j)$-entry of $A \cdot B$ is
    \[
        c_{ij} = \sum_{\ell = 1}^{n} a_{i\ell} b_{\ell j}.
    \]
    Fix $i$ and $j$. A summand $a_{i\ell} b_{\ell j}$ can be non-zero only if $\ell$ lies in the same block as $i$ (else $a_{i\ell} = 0$) \emph{and} in the same block as $j$ (else $b_{\ell j} = 0$). If $i$ and $j$ lie in different blocks, no $\ell$ meets both conditions, the sum is empty of non-zero terms, and $c_{ij} = 0$. This is the assertion that $A \cdot B$ is again block diagonal with blocks of the same two sizes.

    If instead $i$ and $j$ both lie in $I$, the surviving summands are exactly those with $\ell \in I$, so
    \[
        c_{ij} = \sum_{\ell = 1}^{k} a_{i\ell} b_{\ell j},
    \]
    which is the $(i,j)$-entry of $A_1 \cdot B_1$. The same computation over $J$ gives, for $i, j \in J$, the $(i-k, j-k)$-entry of $A_2 \cdot B_2$. Hence
    \[
        A \cdot B = \begin{pmatrix} A_1 B_1 & 0_{k \times (n-k)} \\ 0_{(n-k) \times k} & A_2 B_2 \end{pmatrix},
    \]
    as claimed: a block diagonal product is computed block by block.

    \item Suppose $A_1 \in \GL_k(K)$ and $A_2 \in \GL_{n-k}(K)$, and put
    \[
        C := \begin{pmatrix} A_1^{-1} & 0_{k \times (n-k)} \\ 0_{(n-k) \times k} & A_2^{-1} \end{pmatrix} \in \M_{n \times n}(K),
    \]
    which is block diagonal with the same two block sizes. By part \textbf{(a)},
    \[
        A \cdot C = \begin{pmatrix} A_1 A_1^{-1} & 0 \\ 0 & A_2 A_2^{-1} \end{pmatrix} = \begin{pmatrix} I_k & 0 \\ 0 & I_{n-k} \end{pmatrix} = I_n,
    \]
    and symmetrically $C \cdot A = I_n$. So, $A$ is invertible by \cref{def:invertible_matrix}, and $C$ is its inverse by the uniqueness on \cpageref{prop:uniqueness_of_inverse_matrix}:
    \[
        A^{-1} = \begin{pmatrix} A_1^{-1} & 0_{k \times (n-k)} \\ 0_{(n-k) \times k} & A_2^{-1} \end{pmatrix}.
    \]
    The inverse of a block diagonal matrix is thus obtained by inverting each block separately.

    \item This is exactly the upper triangular case of \cref{lemma:triangular_multiplication_closure}, whose proof is carried out on \cpageref{sol:triangular_multiplication_closure}. In brief: if $A = (a_{ij})$ and $B = (b_{ij})$ are both upper triangular, then $a_{i\ell} = 0$ unless $i \le \ell$ and $b_{\ell j} = 0$ unless $\ell \le j$, so a summand of $\sum_{\ell} a_{i\ell} b_{\ell j}$ survives only when $i \le \ell \le j$. For $i > j$ there is no such $\ell$, and therefore $(AB)_{ij} = 0$. Hence $AB$ is upper triangular, and $U$ is closed under matrix multiplication.

    Note that $U$ is a subspace of $\M_{n \times n}(K)$ that is moreover closed under the product; together with $I_n \in U$ this makes $U$ a subring of the matrix ring of \cref{rem:matrix_ring}, not merely a subspace.
\end{enumerate}
\exinfo{Official Solution (Problem Sheet 11, Exercise 3). Block diagonal matrices multiply and invert block by block; upper triangular matrices are closed under the product.}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{exc:powers_2x2_nilpotency}]
\label{sol:powers_2x2_nilpotency}
\inlinemetanote{Uses later theory. Part \textbf{(a)} runs on two tools this chapter does not have: the \emph{determinant} and its multiplicativity, from ch.\ 20, and the $2 \times 2$ case of the Cayley--Hamilton identity $A^2 - \Tr(A)A + \det(A)I_2 = 0$, from ch.\ 21 (where it is \cref{ex:cayley_2x2}). The identity itself can be checked here by expanding both sides entry by entry, which is what \cref{ex:cayley_2x2} does; it is only the name and the context that arrive later.}

\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $A \in \M_{2 \times 2}(K)$ such that $A^2 \ne 0$.
    Suppose for contradiction that $A^k = 0$ for some $k \ge 3$.
    Taking determinants, $\det(A^k) = (\det A)^k = 0$, which implies $\det A = 0$.
    For any $2 \times 2$ matrix $A$, direct expansion verifies the identity $A^2 - \Tr(A)A + \det(A)I_2 = 0$.
    Since $\det A = 0$, this simplifies to:
    \[
    A^2 = \Tr(A) A.
    \]
    Multiplying repeatedly by $A$ yields $A^m = (\Tr A)^{m-1} A$ for all $m \ge 1$.
    If $\Tr(A) = 0$, then $A^2 = 0 \cdot A = 0$, which contradicts $A^2 \ne 0$.
    If $\Tr(A) \ne 0$, then $A^k = (\Tr A)^{k-1} A = 0 \implies A = 0$, which implies $A^2 = 0$, again a contradiction.
    Therefore, $A^k \ne 0$ for all integers $k \ge 3$.

    \item For any field $K$, consider the non-zero strictly upper triangular matrix:
    \[
    A = \begin{pmatrix} 0 & 1 \\ 0 & 0 \end{pmatrix} \in \M_{2 \times 2}(K) \setminus \{0\}.
    \]
    Direct calculation gives $A^2 = \left(\begin{smallmatrix} 0 & 1 \\ 0 & 0 \end{smallmatrix}\right) \left(\begin{smallmatrix} 0 & 1 \\ 0 & 0 \end{smallmatrix}\right) = \left(\begin{smallmatrix} 0 & 0 \\ 0 & 0 \end{smallmatrix}\right) = 0$.
    Thus $A^n = 0$ for $n = 2$. \qedhere
\end{enumerate}
\exinfo{Official Solution (Problem Sheet 12, Exercise 5). Demonstrates that a 2x2 matrix with non-zero square cannot be nilpotent, and gives a nilpotency index 2 example.}
\end{exercisesolution}

\begin{exercisesolution}[Solution to \cref{exc:one_sided_inverse_finite_dim}]
\label{sol:one_sided_inverse_finite_dim}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $V, W$ be finite-dimensional vector spaces with $\dim V = \dim W = n$, and let $S \in \Hom(V, W)$ and $T \in \Hom(W, V)$ satisfy $T \circ S = \id_V$.
    \begin{itemize}
        \item \emph{$S$ is injective:} If $v \in \ker(S)$, then $S(v) = 0$, so $v = \id_V(v) = (T \circ S)(v) = T(S(v)) = T(0) = 0$. Thus, $\ker(S) = \{0\}$.
        \item \emph{$S$ is an isomorphism:} By \cref{cor:equal_dim_iso} \textbf{(c)} (a consequence of the Rank-Nullity Theorem \cref{thm:rank_nullity}), for a linear map between vector spaces of equal finite dimension $\dim V = \dim W = n < \infty$, injectivity is equivalent to bijectivity. Thus, $S \colon V \to W$ is an isomorphism.
        \item \emph{$T = S^{-1}$:} Since $S$ is invertible, we multiply $T \circ S = \id_V$ on the right by $S^{-1}$ to obtain $T = T \circ (S \circ S^{-1}) = (T \circ S) \circ S^{-1} = \id_V \circ S^{-1} = S^{-1}$.
    \end{itemize}

    \item Let $A, B \in \M_{n \times n}(K)$ such that $AB = I_n$.
    Consider the endomorphisms $T_A, T_B \in \End(K^n)$ defined by $T_A(v) = Av$ and $T_B(v) = Bv$.
    The equation $AB = I_n$ implies $T_A \circ T_B = \id_{K^n}$.
    By part \textbf{(a)} (with $V = W = K^n$), $T_B$ is an isomorphism with $T_B^{-1} = T_A$.
    Hence $T_B \circ T_A = \id_{K^n}$, which means $BA = I_n$.
    Thus $A$ is invertible with inverse $A^{-1} = B$. \qedhere
\end{enumerate}
\exinfo{Official Solution (Problem Sheet 13, Exercise 3). Proves that a one-sided inverse of a finite-dimensional endomorphism or square matrix is a full two-sided inverse.}
\end{exercisesolution}

\begin{exercisesolution*}[Solution to \cref{exc:gl_not_subspace_sc}]
\label{sol:gl_not_subspace_sc}
The correct answer is \textbf{(a): not a real linear subspace of $\M_{n \times n}(\mathbb{R})$}.

\textbf{Justification:} A linear subspace must contain the zero element of the ambient vector space.
However, the zero matrix $0 \in \M_{n \times n}(\mathbb{R})$ has determinant $\det(0) = 0$, so $0 \notin \GL_n(\mathbb{R})$.
Furthermore, $\GL_n(\mathbb{R})$ is not closed under addition: for instance, $I_n \in \GL_n(\mathbb{R})$ and $-I_n \in \GL_n(\mathbb{R})$, but their sum $I_n + (-I_n) = 0 \notin \GL_n(\mathbb{R})$. \qedhere
\exinfo{Official Solution (Problem Sheet 14, Single Choice Question 3). Shows that the general linear group $\GL_n(\mathbb{R})$ is not a subspace because it lacks the zero matrix.}
\end{exercisesolution*}
