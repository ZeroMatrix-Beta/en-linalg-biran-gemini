\chapter{Dual Spaces and Inner Products}

\section{Dual spaces and inner products}

Let $V$ be a vector space over $K$ (any field). Recall that we have the dual space $V^* := \Hom(V, K)$, which is the space of linear maps $\varphi \colon V \to K$ (also called \newterm{functionals}). The dual space $V^*$ is also a vector space over $K$.

\begin{warmup}
Assume $V = K^n$ and let $\varphi \colon K^n \to K$ be a functional. Then there exists a unique matrix $A = (a_1, \dots, a_n) \in \M_{1 \times n}(K)$ such that $\varphi(v) = A \cdot v$ for all $v \in K^n$.

Indeed, let $e_1, \dots, e_n$ be the standard basis for $K^n$, and define $a_1 := \varphi(e_1), \dots, a_n := \varphi(e_n)$. Let $v = (v_1, \dots, v_n)\transp \in K^n$. Then
\[
    \varphi(v) = \varphi(v_1 e_1 + \dots + v_n e_n) = v_1 \varphi(e_1) + \dots + v_n \varphi(e_n) = v_1 a_1 + \dots + v_n a_n = A \cdot v.
\]
% Creator: Opus 5
This also settles the uniqueness of $A$: any row matrix $A' = (a_1', \dots, a_n')$ with $\varphi(v) = A' \cdot v$ for all $v \in K^n$ satisfies $a_i' = A' \cdot e_i = \varphi(e_i) = a_i$, so its entries are already forced by $\varphi$.
% Extractor: Gemini 3.1 Pro
\end{warmup}

Consider now an inner product space $(V, \langle \cdot, \cdot \rangle)$ over $K$ (where $K = \mathbb{R}$ or $\mathbb{C}$). Let $u \in V$. We define a map $\varphi_u \colon V \to K$ by mapping $v \mapsto \langle v, u \rangle \in K$.

\begin{claim*}
\label{claim:linearity_phi_u}
The map $\varphi_u$ is linear.
\end{claim*}
(This is left as an exercise).

% Thm 23.a.1
\begin{theorem}
\label{thm:dual_space_inner_product_isomorphism}
Let $(V, \langle \cdot, \cdot \rangle)$ be a finite-di\-men\-sional inner product space over $K$ (where $K = \mathbb{R}$ or $\mathbb{C}$). Let $\varphi \in V^*$. Then there exists a unique vector $u \in V$ such that $\varphi = \varphi_u$, that is, $\varphi(v) = \langle v, u \rangle$ for all $v \in V$.

In fact, for $K = \mathbb{R}$, the map $\Phi \colon V \to V^*$, given by $\Phi(u) := \varphi_u$, is an isomorphism. For $K = \mathbb{C}$, this map is injective and surjective, but it is only $\mathbb{C}$-antilinear, meaning that $\Phi(u_1+u_2) = \Phi(u_1)+\Phi(u_2)$ and $\Phi(c \cdot u) = \overline{c} \cdot \Phi(u)$ for all $c \in \mathbb{C}$ and $u \in V$.

So, at least for $K = \mathbb{R}$, a scalar product on $V$ gives us a canonical isomorphism $V \cong V^*$. (However, this isomorphism depends on the scalar product!)
\end{theorem}

\begin{proof}
First, we prove that for all $\varphi \in V^*$ there exists a unique $u \in V$ such that $\varphi = \varphi_u$.
Fix an orthonormal basis $\mathcal{B} := (e_1, \dots, e_n)$ for $V$, where $n = \dim V$; one exists by \cref{cor:orthonormal_basis_exists}.
Let $\varphi \in V^*$, and let $v \in V$. We have $v = \sum_{i=1}^n \langle v, e_i \rangle \cdot e_i$, the coefficients of a vector in an orthonormal system being given by \cref{thm:properties_orthogonal_systems} \textbf{(b)}, with $\langle e_i, e_i \rangle = 1$. This implies:
\[
    \varphi(v) = \varphi\left( \sum_{i=1}^n \langle v, e_i \rangle \cdot e_i \right) = \sum_{i=1}^n \langle v, e_i \rangle \cdot \varphi(e_i) = \left\langle v, \sum_{i=1}^n \overline{\varphi(e_i)} \cdot e_i \right\rangle.
\]
(Note that the conjugation is needed here only if $K = \mathbb{C}$).
Define $u := \sum_{i=1}^n \overline{\varphi(e_i)} \cdot e_i$. It follows that $\varphi(v) = \langle v, u \rangle$, so $\varphi = \varphi_u$.

\textbf{Uniqueness of $u$:} Suppose that $\varphi_{u_1} = \varphi_{u_2}$ for $u_1, u_2 \in V$. This implies that $\langle v, u_1 \rangle = \langle v, u_2 \rangle$ for all $v \in V$. By a previous result (\cref{lem:basic_properties_inner_prod} \textbf{(c)}), this implies that $u_1 = u_2$.

The statements about the linearity of $\Phi$ are left as an exercise.
\end{proof}

\begin{exercise}[Dual Inner Product and Isometry via Riesz Map]
  \label{exc:dual_space_inner_product_isometry}
  Let $(V, \langle \cdot, \cdot \rangle)$ be a finite-dimensional Euclidean vector space, and consider the isomorphism
  \[
  \delta \colon V \longrightarrow V^* := \Hom(V, \mathbb{R}), \qquad v \longmapsto \delta(v) := \langle v, \cdot \rangle.
  \]
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item Show that there exists a unique scalar product $\langle \cdot, \cdot \rangle^*$ on $V^*$ such that $\delta$ is an isometry, i.e., $\langle \delta(v), \delta(w) \rangle^* = \langle v, w \rangle$ for all $v, w \in V$.
    \item Let $\mathcal{B} = (v_1, \dots, v_n)$ be an ordered basis of $V$, and let $\mathcal{B}^* = (v_1^*, \dots, v_n^*)$ be the corresponding dual basis of $V^*$. Express the representation matrix of $\langle \cdot, \cdot \rangle^*$ with respect to $\mathcal{B}^*$ in terms of the representation matrix $G = [\langle \cdot, \cdot \rangle]_{\mathcal{B}} = (\langle v_i, v_j \rangle)_{i,j=1}^n$.
  \end{enumerate}
  \exhint[Official Hint]{(a) Since $\delta$ is a vector space isomorphism, the isometry requirement forces $\langle \varphi, \psi \rangle^* := \langle \delta^{-1}(\varphi), \delta^{-1}(\psi) \rangle$ for all $\varphi, \psi \in V^*$. Check that this formula indeed satisfies all axioms of a scalar product (bilinearity, symmetry, and positivity) using the linearity and injectivity of $\delta^{-1}$.
  (b) The basis change from the dual basis $\mathcal{B}^*$ to the basis $\delta(\mathcal{B}) = (\delta(v_1), \dots, \delta(v_n))$ of $V^*$ is given by the Gram matrix $G$: since $v_i^* = \sum_{k=1}^n (G^{-1})_{ik} \delta(v_k)$, it follows that the representation matrix of $\langle \cdot, \cdot \rangle^*$ with respect to $\mathcal{B}^*$ is $[\langle \cdot, \cdot \rangle^*]_{\mathcal{B}^*} = G^{-1}$.}
  \exinfo{This exercise is Problem 6 of Problem Sheet 23, Linear Algebra II, Spring Semester 2026, ETH Zurich (Lecturer: Prof. P. Biran).}
\exsol{sol:dual_space_inner_product_isometry}
\end{exercise}

\begin{exercise}[Failure of Riesz Representation in Infinite Dimensions]
  \label{exc:riesz_failure_infinite_dimensions}
  Let $V = C[-1, 1]$ be the vector space of real continuous functions equipped with the inner product $\langle f, g \rangle := \int_{-1}^1 f(x) g(x) \, dx$. Consider the linear functional
  \[
  \varphi \colon V \longrightarrow \mathbb{R}, \qquad f \longmapsto f(0).
  \]
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item Show that there does not exist any function $g \in V$ such that $\varphi(f) = \langle f, g \rangle$ for all $f \in V$.
    \item Explain why this does not contradict \cref{thm:dual_space_inner_product_isomorphism}.
  \end{enumerate}
  \exhint[Official Hint]{(a) Suppose such a continuous $g \in C[-1, 1]$ exists. For each $n \in \mathbb{N}$, consider the continuous tent functions $f_n(x) = \max\{0, 1 - n|x|\}$. Then $\varphi(f_n) = f_n(0) = 1$, but $\langle f_n, g \rangle = \int_{-1/n}^{1/n} f_n(x) g(x) \, dx \to 0$ as $n \to \infty$ because $|g(x)| \le M$ is bounded. This yields $1 = 0$, a contradiction.
  (b) \Cref{thm:dual_space_inner_product_isomorphism} assumes that $\dim V < \infty$, whereas $C[-1, 1]$ is infinite-dimensional.}
  \exinfo{This exercise is Problem 5 of Problem Sheet 24, Linear Algebra II, Spring Semester 2026, ETH Zurich (Lecturer: Prof. P. Biran).}
\exsol{sol:riesz_failure_infinite_dimensions}
\end{exercise}

\begin{exercise}[Existence of the Adjoint for Euclidean Homomorphisms]
  \label{exc:adjoint_existence_finite_dim}
  Let $f \colon V \to W$ be a linear map between Euclidean vector spaces.
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item Assume $\dim V < \infty$. Show that there exists a unique linear map $f^* \colon W \to V$ such that
    \[
    \langle f(v), w \rangle_W = \langle v, f^*(w) \rangle_V \qquad \text{for all } v \in V, \, w \in W.
    \]
    \item Does the statement in \textbf{(a)} always hold if we instead assume $\dim W < \infty$ while $\dim V = \infty$?
  \end{enumerate}
  \exhint[Official Hint]{(a) For each fixed $w \in W$, the map $\varphi_w \colon V \to \mathbb{R}$ given by $\varphi_w(v) = \langle f(v), w \rangle_W$ is a linear functional on $V$. Since $\dim V < \infty$, the Riesz representation theorem (\cref{thm:dual_space_inner_product_isomorphism}) guarantees a unique vector $v_w \in V$ such that $\varphi_w(v) = \langle v, v_w \rangle_V$. Defining $f^*(w) := v_w$ gives the required map. Show that $f^*$ is linear and unique.
  (b) No. Consider $V = \mathbb{R}^{(\mathbb{N})}$ (the space of real sequences $(a_i)_{i=1}^\infty$ with only finitely many non-zero terms) equipped with the standard $\ell^2$-scalar product $\langle a, b \rangle = \sum_{i=1}^\infty a_i b_i$, and let $W = \mathbb{R}$. Define $f \colon V \to \mathbb{R}$ by $f(a) = \sum_{i=1}^\infty a_i$. If $f^*$ existed, $f^*(1)$ would be a vector $c \in V$ such that $f(a) = \langle a, c \rangle = \sum a_i c_i$. Taking $a = e_k$ gives $c_k = f(e_k) = 1$ for all $k \in \mathbb{N}$, which has infinitely many non-zero entries and thus does not belong to $V$.}
  \exinfo{This exercise is Problem 5 of Problem Sheet 25, Linear Algebra II, Spring Semester 2026, ETH Zurich (Lecturer: Prof. P. Biran).}
\exsol{sol:adjoint_existence_finite_dim}
\end{exercise}

\subsection*{The adjoint map}

Let $V, W$ be vector spaces over $K$. Let $T \colon V \to W$ be a linear map. We have seen in Linear Algebra I that $T$ induces a linear map $T^* \colon W^* \to V^*$ defined by $T^*(\varphi) := \varphi \circ T$.

\begin{center}
\begin{tikzcd}
V \arrow[r, "T"] \arrow[dr, "T^*(\varphi) = \varphi \circ T"'] & W \arrow[d, "\varphi"] \\
& K
\end{tikzcd}
\end{center}

Assume now that $(V, \langle \cdot, \cdot \rangle_V)$ and $(W, \langle \cdot, \cdot \rangle_W)$ are inner product spaces over $K$ (where $K = \mathbb{R}$ or $K = \mathbb{C}$). Assume that $\dim V < \infty$.
Given $T$ as above, we can define a map $T' \colon W \to V$ using our \qt{canonical} identifications $W \cong W^*$ and $V \cong V^*$. (This is straightforward for $K = \mathbb{R}$, but we have to be careful with $K = \mathbb{C}$.)

% The notes draw two parallel arrows between $V$ and $V^*$: $\Phi_V$ pointing up
% and $\Phi_V^{-1}$ pointing down. Both variants of the transcription drew
% $\Phi_V^{-1}$ from $V$ to $V^*$, which is the wrong direction for the inverse.
\begin{center}
\begin{tikzcd}[row sep=large, column sep=large]
W^* \arrow[r, "T^*"] & V^* \arrow[d, shift left=1.1ex, "\Phi_V^{-1}"] \\
W \arrow[u, "\Phi_W"] \arrow[r, dashed, "T'"'] & V \arrow[u, shift left=1.1ex, "\Phi_V"]
\end{tikzcd}
\end{center}

Here, we define $T' := \Phi_V^{-1} \circ T^* \circ \Phi_W$. (Note that here we are using the fact that $\dim V < \infty$ so that $\Phi_V^{-1}$ exists).

What properties does $T'$ have? Note that $T'$ is uniquely defined by $\Phi_V \circ T' = T^* \circ \Phi_W$. We can unpack this relation as follows:
\begin{align*}
    \Phi_V \circ T' &= T^* \circ \Phi_W \\
    &\iff \forall w \in W, \quad (\Phi_V \circ T')(w) = (T^* \circ \Phi_W)(w) \\
    &\iff \varphi_{T'(w)} = T^*(\varphi_w) \quad (\text{in } V^*) \\
    &\iff \forall v \in V, \quad \varphi_{T'(w)}(v) = (T^*(\varphi_w))(v) \\
    &\iff \langle v, T'(w) \rangle_V = \varphi_w(T(v)) = \langle T(v), w \rangle_W.
\end{align*}
In other words,
\begin{equation}
\label{eq:adjoint_property}
    \langle T(v), w \rangle_W = \langle v, T'(w) \rangle_V \quad \forall v \in V, \forall w \in W.
\end{equation}

\begin{claim*}
The map $T'$ is a linear map.
\end{claim*}

\begin{proof}
For $K = \mathbb{R}$, the maps $\Phi_W$, $\Phi_V$, and $T^*$ are linear, and therefore $T'$ is also linear.
For $K = \mathbb{C}$, the maps $\Phi_W$ and $\Phi_V$ are additive but only $\mathbb{C}$-antilinear. This implies that $\Phi_V^{-1}$ is also additive and $\mathbb{C}$-antilinear (this is an exercise to check!).
The map $T^*$ remains linear.
Consequently, $T'(w_1 + w_2) = T'(w_1) + T'(w_2)$ (exercise).\label{exc:additivity_T_prime}
Now, let $c \in \mathbb{C}$ and $w \in W$. We compute:
\begin{align*}
    T'(cw) &= (\Phi_V^{-1} \circ T^* \circ \Phi_W)(cw) \\
    &= \Phi_V^{-1}(T^*(\overline{c} \cdot \Phi_W(w))) \\
    &= \Phi_V^{-1}(\overline{c} \cdot T^*(\Phi_W(w))) \\
    &= c \cdot \Phi_V^{-1}(T^*(\Phi_W(w))) \\
    &= c \cdot T'(w).
\end{align*}
This proves that $T'$ is linear.
\end{proof}

\begin{notation}
The linear map $T' \colon W \to V$ is denoted many times by $T^* \colon W \to V$. This makes sense because of the \qt{canonical} identifications:
\begin{center}
\begin{tikzcd}[row sep=large, column sep=large]
T \colon V \to W \quad \rightsquigarrow \quad & W^* \arrow[r, "T^*"] & V^* \\
& W \arrow[u, "\cong" sloped, draw=none] \arrow[r, "T'"'] & V \arrow[u, "\cong" sloped, draw=none]
\end{tikzcd}
\end{center}
(Here we also assume $\dim W < \infty$ to have the canonical identification on the left). The map $T^* \colon W \to V$ is called the \newterm{adjoint} of $T$.
\end{notation}

\subsection*{Basic properties of adjoint maps}

% Lemma 23.a.2
\begin{lemma}[Properties of the Adjoint Map]
\label{lem:properties_adjoint_map}
Let $V, W, U$ be inner product spaces over $K$ (where $K = \mathbb{R}$ or $\mathbb{C}$), all finite-di\-men\-sional. Let $S, T \colon V \to W$ and $R \colon W \to U$ be linear maps. Then:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $T^*$ is a linear map.
    \item $(S+T)^* = S^* + T^*$.
    \item $(\lambda T)^* = \overline{\lambda} T^*$ for all $\lambda \in K$.
    \item $(T^*)^* = T$.
    \item $(\id_V)^* = \id_V$.
    \item $(R \circ T)^* = T^* \circ R^*$.
\end{enumerate}
\end{lemma}

\begin{proof}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item This was proved already. (We will use repeatedly that if $\langle x, y_1 \rangle = \langle x, y_2 \rangle$ for all $x$, then $y_1 = y_2$; this is \cref{lem:basic_properties_inner_prod} \textbf{(c)}).
    \item Let $w \in W$ and $v \in V$. We have:
    \begin{align*}
        \langle v, (S+T)^*(w) \rangle &= \langle (S+T)(v), w \rangle \\
        &= \langle S(v), w \rangle + \langle T(v), w \rangle \\
        &= \langle v, S^*(w) \rangle + \langle v, T^*(w) \rangle \\
        &= \langle v, (S^*+T^*)(w) \rangle.
    \end{align*}
    This holds for all $v \in V$, which implies $(S+T)^*(w) = (S^*+T^*)(w)$. Since this holds for all $w \in W$, we conclude that $(S+T)^* = S^* + T^*$.
    \item This is left as an exercise.
    \item Let $v \in V$ and $w \in W$. We have:
    \begin{align*}
        \langle w, (T^*)^*(v) \rangle &= \langle T^*(w), v \rangle \\
        &= \overline{\langle v, T^*(w) \rangle} \\
        &= \overline{\langle T(v), w \rangle} \\
        &= \langle w, T(v) \rangle.
    \end{align*}
    This holds for all $w \in W$, which implies $(T^*)^*(v) = T(v)$. Since this holds for all $v \in V$, we conclude that $(T^*)^* = T$.
    \item This is left as an exercise.
    \item Let $v \in V$ and $u \in U$. We have:
    \begin{align*}
        \langle v, (R \circ T)^*(u) \rangle &= \langle (R \circ T)(v), u \rangle \\
        &= \langle R(T(v)), u \rangle \\
        &= \langle T(v), R^*(u) \rangle \\
        &= \langle v, T^*(R^*(u)) \rangle \\
        &= \langle v, (T^* \circ R^*)(u) \rangle.
    \end{align*}
    This holds for all $v \in V$, which implies $(R \circ T)^*(u) = (T^* \circ R^*)(u)$. Since this holds for all $u \in U$, we conclude that $(R \circ T)^* = T^* \circ R^*$.
\end{enumerate}
\end{proof}

% Lemma 23.a.3
\begin{lemma}[Orthogonal Relations of Adjoint Maps]
\label{lem:orthogonal_relations_adjoint_maps}
Let $V, W$ be finite-di\-men\-sional inner product spaces. Let $T \colon V \to W$ and let $T^* \colon W \to V$ be the adjoint of $T$. Then:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $\ker(T^*) = (\Img T)^\perp$.
    \item $\Img T^* = (\ker T)^\perp$.
    \item $\ker T = (\Img T^*)^\perp$.
    \item $\Img T = (\ker T^*)^\perp$.
\end{enumerate}
\end{lemma}

% Extractor: Opus 5
\begin{proof}
Only \textbf{(a)} is established from scratch; the notes then deduce the other three from it in the order \textbf{(d)}, \textbf{(b)}, \textbf{(c)}, each step relying only on the ones already obtained. The forward references below are therefore not circular. The passages from a subspace to its double orthogonal complement all invoke \cref{cor:bidual_orthogonal_complement}, which requires the subspace in question to be finite-di\-men\-sional; we point out each time why it is.
% Extractor: Gemini 3.1 Pro
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $w \in W$. Then:
    \begin{align*}
        w \in \ker(T^*) &\iff T^*(w) = 0 \\
        &\iff \forall v \in V, \quad \langle v, T^*(w) \rangle = 0 \\
        &\iff \forall v \in V, \quad \langle T(v), w \rangle = 0 \\
        &\iff w \perp T(v) \quad \forall v \in V \\
        &\iff w \perp \Img(T) \\
        &\iff w \in (\Img T)^\perp.
    \end{align*}

    \item Use statement \textbf{(d)} with $T^*$ instead of $T$:
    \[
        \Img T^* = (\ker (T^*)^*)^\perp = (\ker T)^\perp.
    \]

    \item Apply the orthogonal complement $(-)^\perp$ to statement \textbf{(b)}:
    \[
        (\Img T^*)^\perp = ((\ker T)^\perp)^\perp = \ker T.
    \]
% Extractor: Opus 5
    The second equality is \cref{cor:bidual_orthogonal_complement} applied to the subspace $\ker T \subseteq V$, which is finite-di\-men\-sional because $V$ is.
% Extractor: Gemini 3.1 Pro

    \item Apply the orthogonal complement $(-)^\perp$ to both sides of \textbf{(a)}:
    \[
        (\ker T^*)^\perp = ((\Img T)^\perp)^\perp = \Img T.
    \]
% Extractor: Opus 5
    Here the second equality is \cref{cor:bidual_orthogonal_complement} applied to $\Img T \subseteq W$; recall that $\Img T$ is finite-di\-men\-sional, being the image of the finite-di\-men\-sional space $V$.
% Extractor: Gemini 3.7 Flash (High)
\end{enumerate}
\end{proof}

\begin{exercise}[Combinatorial Gradient, Adjoint Divergence, and Graph Laplacian]
  \label{exc:graph_laplacian_adjoint}
  Let $G = (V, E)$ be a finite directed graph with vertices $V$ and directed edges $E \subseteq \{(v_{\text{init}}, v_{\text{term}}) \in V \times V \mid v_{\text{init}} \neq v_{\text{term}}\}$. Consider the real function spaces $\mathbb{R}^V = \{f \colon V \to \mathbb{R}\}$ and $\mathbb{R}^E = \{\varphi \colon E \to \mathbb{R}\}$ equipped with the standard inner products
  \[
  \langle f_1, f_2 \rangle_V := \sum_{v \in V} f_1(v) f_2(v), \qquad \langle \varphi_1, \varphi_2 \rangle_E := \sum_{e \in E} \varphi_1(e) \varphi_2(e).
  \]
  Define the \newterm{combinatorial derivative} (gradient) $T \colon \mathbb{R}^V \to \mathbb{R}^E$ by $(Tf)(e) := f(v_{\text{term}}) - f(v_{\text{init}})$ for $e = (v_{\text{init}}, v_{\text{term}})$, and define $S \colon \mathbb{R}^E \to \mathbb{R}^V$ by
  \[
  (S\varphi)(v) := \sum_{\substack{v_{\text{init}} \in V \\ (v_{\text{init}}, v) \in E}} \varphi(v_{\text{init}}, v) - \sum_{\substack{v_{\text{term}} \in V \\ (v, v_{\text{term}}) \in E}} \varphi(v, v_{\text{term}}).
  \]
  \begin{enumerate}[label=\textbf{(\alph*)}]
    \item Show that $T^* = S$, and compute the composite $\Delta := T^* \circ T = S \circ T \colon \mathbb{R}^V \to \mathbb{R}^V$ (the \newterm{combinatorial Laplace operator} of $G$).
    \item Assume that $G$ is undirected (i.e., $(v, w) \in E \iff (w, v) \in E$) and $d$-regular (each vertex has degree $d$). Show that $\Delta = T^* T$ has eigenvalue $0$, and explain why the geometric multiplicity $m_g(0)$ equals the number of connected components of $G$.
  \end{enumerate}
  \exhint[Official Hint]{(a) By definition of the adjoint, show $\langle Tf, \varphi \rangle_E = \langle f, S\varphi \rangle_V$. Expanding both sums over $e = (v_{\text{init}}, v_{\text{term}}) \in E$ yields $\sum_{e \in E} (f(v_{\text{term}}) - f(v_{\text{init}})) \varphi(e)$, which matches $\sum_{v \in V} f(v) (S\varphi)(v)$.
  (b) Note that $\Delta f = 0 \iff \langle \Delta f, f \rangle_V = \|Tf\|_E^2 = 0 \iff Tf = 0 \iff f(v_{\text{term}}) = f(v_{\text{init}})$ for all $(v_{\text{init}}, v_{\text{term}}) \in E$. Thus, $f \in \ker(\Delta)$ if and only if $f$ is constant on every connected component of $G$. The dimension of $\ker(\Delta) = \gEig_\Delta(0)$ (the geometric multiplicity) is therefore exactly the number of connected components.}
  \exinfo{This exercise is Problem 6 of Problem Sheet 24, Linear Algebra II, Spring Semester 2026, ETH Zurich (Lecturer: Prof. P. Biran).}
\exsol{sol:graph_laplacian_adjoint}
\end{exercise}

\begin{exercise}[Invariant Subspaces and the Adjoint Map \faHeart]
  \label{exc:invariant_subspaces_adjoint}
  Let $V$ be a finite-dimensional inner product space over $K \in \{\mathbb{R}, \mathbb{C}\}$, let $T \in \End(V)$, and let $U \subseteq V$ be a subspace. Prove that $U$ is $T$-invariant if and only if $U^\perp$ is $T^*$-invariant.
  \exhint[Hint (Y. St\"aubli)]{Suppose $T(U) \subseteq U$. Let $w \in U^\perp$. We must show $T^*(w) \in U^\perp$, meaning $\langle u, T^*(w) \rangle = 0$ for all $u \in U$. By definition of the adjoint, $\langle u, T^*(w) \rangle = \langle T(u), w \rangle$. Since $T(u) \in U$ and $w \in U^\perp$, this inner product is zero, so $T^*(w) \in U^\perp$.
  Conversely, if $U^\perp$ is $T^*$-invariant, applying the forward direction to $T^*$ and $U^\perp$ gives that $(U^\perp)^\perp$ is $(T^*)^*$-invariant. Since $(U^\perp)^\perp = U$ (using $\dim V < \infty$, \cref{cor:bidual_orthogonal_complement}) and $(T^*)^* = T$, this shows $T(U) \subseteq U$.}
  \exinfo{This exercise is Problem 1 of Problem Sheet 25, Linear Algebra II, Spring Semester 2026, ETH Zurich (Lecturer: Prof. P. Biran).}
\exsol{sol:invariant_subspaces_adjoint}
\end{exercise}

\subsection*{Matrix representation of adjoint maps}

% Prop 23.a.4
\begin{proposition}[Matrix Representation of the Adjoint]
\label{prop:matrix_representation_adjoint}
Let $V, W$ be finite-di\-men\-sional inner product spaces and $T \colon V \to W$ a linear map. Let $\mathcal{B} := (e_1, \dots, e_n)$ be an orthonormal basis for $V$, and $\mathcal{C} := (f_1, \dots, f_m)$ an orthonormal basis for $W$. Then:
\[
    [T^*]_{\mathcal{B}}^{\mathcal{C}} = \left( \overline{[T]_{\mathcal{C}}^{\mathcal{B}}} \right)\transp = \left([T]_{\mathcal{C}}^{\mathcal{B}}\right)^* \quad \text{(\newterm{adjoint matrix})}
\]
\end{proposition}

\begin{proof}
Write $A := (a_{ij})_{\substack{1 \leq i \leq m \\ 1 \leq j \leq n}} := [T]_{\mathcal{C}}^{\mathcal{B}}$ and $B := (b_{ij})_{\substack{1 \leq i \leq n \\ 1 \leq j \leq m}} := [T^*]_{\mathcal{B}}^{\mathcal{C}}$.
Both bases being orthonormal, \cref{prop:matrix_representation_orthonormal} reads the entries of a representation matrix off the inner product: $b_{ij} = \langle T^*(f_j), e_i \rangle$ and $a_{ji} = \langle T(e_i), f_j \rangle$. Hence
\[
    b_{ij} = \langle T^*(f_j), e_i \rangle = \overline{\langle e_i, T^*(f_j) \rangle} = \overline{\langle T(e_i), f_j \rangle} = \overline{a_{ji}},
\]
the middle step by the Hermitian axiom and the next by the defining property \eqref{eq:adjoint_property} of the adjoint.
Therefore, $B = (\overline{A})\transp = A^*$.
\end{proof}

% Cor 23.a.5
\begin{corollary}[Adjoint of a Matrix Map]
\label{cor:adjoint_matrix_map}
Let $A \in \M_{m \times n}(K)$ (where $K = \mathbb{R}$ or $\mathbb{C}$), and let $T_A \colon K^n \to K^m$ be the linear map defined by $T_A(v) = A \cdot v$. Endow $K^n$ and $K^m$ with their standard inner products. Then $(T_A)^* = T_{A^*}$.
\end{corollary}

% Creator: Opus 5
\begin{proof}
Let $\mathcal{E} := (e_1, \dots, e_n)$ and $\mathcal{F} := (f_1, \dots, f_m)$ be the standard bases of $K^n$ and of $K^m$. With respect to the standard inner products both are orthonormal, so \cref{prop:matrix_representation_adjoint} applies to them. The $j$-th column of $[T_A]_{\mathcal{F}}^{\mathcal{E}}$ is $T_A(e_j) = A \cdot e_j$, the $j$-th column of $A$, and therefore $[T_A]_{\mathcal{F}}^{\mathcal{E}} = A$. Consequently,
\[
    [(T_A)^*]_{\mathcal{E}}^{\mathcal{F}} = \left( [T_A]_{\mathcal{F}}^{\mathcal{E}} \right)^* = A^*.
\]
The same computation applied to $A^*$ gives $[T_{A^*}]_{\mathcal{E}}^{\mathcal{F}} = A^*$. Two linear maps $K^m \to K^n$ with the same representation matrix with respect to the same pair of bases are equal, and hence $(T_A)^* = T_{A^*}$.
\end{proof}

\begin{proposition}[Properties of the Adjoint Matrix]
\label{prop:properties_adjoint_matrix}
Let $K = \mathbb{R}, \mathbb{C}$, and let $A, B \in \M_{m \times n}(K)$ and $C \in \M_{n \times p}(K)$. Then:
\begin{enumerate}[label=\textbf{(\alph*)}]

\end{enumerate}
\end{proposition}

\subsection{Solutions to the Exercises}

\begin{exercisesolution}[Proof of Linearity of $\varphi_u$ (on \cpageref{claim:linearity_phi_u})]
\label{sol:linearity_phi_u}
Let $v_1, v_2 \in V$ and $c \in K$. We have:
\[
    \varphi_u(v_1 + v_2) = \langle v_1 + v_2, u \rangle = \langle v_1, u \rangle + \langle v_2, u \rangle = \varphi_u(v_1) + \varphi_u(v_2),
\]
and
\[
    \varphi_u(c v_1) = \langle c v_1, u \rangle = c \langle v_1, u \rangle = c \varphi_u(v_1).
\]
Thus, $\varphi_u$ is a linear map.
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{thm:dual_space_inner_product_isomorphism}]
\label{sol:dual_space_inner_product_isomorphism}
For $K = \mathbb{R}$, we have $\Phi(u) = \varphi_u$. Then:
\[
    \Phi(u_1+u_2)(v) = \langle v, u_1+u_2 \rangle = \langle v, u_1 \rangle + \langle v, u_2 \rangle = \Phi(u_1)(v) + \Phi(u_2)(v),
\]
and
\[
    \Phi(c u)(v) = \langle v, c u \rangle = c \langle v, u \rangle = c \Phi(u)(v).
\]
(Here we used the fact that for $K = \mathbb{R}$, $c = \overline{c}$). Thus, $\Phi$ is linear.

For $K = \mathbb{C}$, the additivity follows identically. For scalar multiplication:
\[
    \Phi(c u)(v) = \langle v, c u \rangle = \overline{c} \langle v, u \rangle = \overline{c} \Phi(u)(v).
\]
Thus, $\Phi(c u) = \overline{c} \Phi(u)$, making it $\mathbb{C}$-antilinear.

For the inverse $\Phi^{-1}$: since $\Phi$ is additive, taking $\varphi_1 = \Phi(u_1)$ and $\varphi_2 = \Phi(u_2)$ gives $\Phi^{-1}(\varphi_1 + \varphi_2) = \Phi^{-1}(\varphi_1) + \Phi^{-1}(\varphi_2)$, so $\Phi^{-1}$ is additive.
For scalar multiplication, let $\varphi = \Phi(u)$ and $\lambda \in \mathbb{C}$. Then $\overline{\lambda} \varphi = \overline{\lambda} \Phi(u) = \Phi(\lambda u)$. Applying $\Phi^{-1}$ to both sides gives $\Phi^{-1}(\overline{\lambda} \varphi) = \lambda u = \lambda \Phi^{-1}(\varphi)$. Setting $c = \overline{\lambda}$ (so $\lambda = \overline{c}$) gives $\Phi^{-1}(c \varphi) = \overline{c} \Phi^{-1}(\varphi)$, proving $\Phi^{-1}$ is $\mathbb{C}$-antilinear.

% Creator: Opus 5
This last paragraph is also the verification Prof. Biran marks with \qt{check!} in the proof that $T'$ is linear: it is exactly the assertion that $\Phi_V^{-1}$ is additive and $\mathbb{C}$-antilinear.
% Creator: Gemini 3.1 Pro
\end{exercisesolution}

\begin{exercisesolution}[Proof of Additivity of $T'$ (on \cpageref{exc:additivity_T_prime})]
\label{sol:additivity_T_prime}
We have:
\begin{align*}
    T'(w_1 + w_2) &= (\Phi_V^{-1} \circ T^* \circ \Phi_W)(w_1 + w_2) \\
    &= (\Phi_V^{-1} \circ T^*)(\Phi_W(w_1) + \Phi_W(w_2)) && \text{(since $\Phi_W$ is additive)} \\
    &= \Phi_V^{-1}(T^*(\Phi_W(w_1)) + T^*(\Phi_W(w_2))) && \text{(since $T^*$ is linear)} \\
    &= \Phi_V^{-1}(T^*(\Phi_W(w_1))) + \Phi_V^{-1}(T^*(\Phi_W(w_2))) && \text{(since $\Phi_V^{-1}$ is additive)} \\
    &= T'(w_1) + T'(w_2).
\end{align*}
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{lem:properties_adjoint_map} \textbf{(c)}]
\label{sol:properties_adjoint_map_c}
Let $v \in V$ and $w \in W$. We have:
\[
    \langle v, (\lambda T)^*(w) \rangle = \langle (\lambda T)(v), w \rangle = \langle \lambda T(v), w \rangle = \lambda \langle T(v), w \rangle = \lambda \langle v, T^*(w) \rangle = \langle v, \overline{\lambda} T^*(w) \rangle.
\]
This holds for all $v \in V$, so $(\lambda T)^*(w) = \overline{\lambda} T^*(w)$. Thus, $(\lambda T)^* = \overline{\lambda} T^*$.
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{lem:properties_adjoint_map} \textbf{(e)}]
\label{sol:properties_adjoint_map_e}
Let $v, w \in V$. We have:
\[
    \langle v, (\id_V)^*(w) \rangle = \langle \id_V(v), w \rangle = \langle v, w \rangle.
\]
This holds for all $v \in V$, so $(\id_V)^*(w) = w = \id_V(w)$. Thus, $(\id_V)^* = \id_V$.
\end{exercisesolution}

\begin{exercisesolution}[Proof of \cref{prop:properties_adjoint_matrix}]
\label{sol:properties_adjoint_matrix}
These statements follow directly from the properties of matrix transpose and complex conjugation:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $\overline{(A+B)}\transp = (\overline{A} + \overline{B})\transp = \overline{A}\transp + \overline{B}\transp$.
    \item $\overline{(\lambda A)}\transp = (\overline{\lambda} \overline{A})\transp = \overline{\lambda} \overline{A}\transp$.
    \item $\overline{(\overline{A}\transp)}\transp = (\overline{\overline{A}}\transp)\transp = (A\transp)\transp = A$.
    \item $\overline{I_n}\transp = I_n\transp = I_n$ (since $I_n$ is a real matrix).
    \item $\overline{(A \cdot C)}\transp = (\overline{A} \cdot \overline{C})\transp = \overline{C}\transp \cdot \overline{A}\transp$.
\end{enumerate}
\end{exercisesolution}
