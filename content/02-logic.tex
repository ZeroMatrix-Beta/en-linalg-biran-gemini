\chapter{Logic and Mathematics}

Before advancing into the formal theory of vector spaces, we must establish the rigorous grammar of mathematical reasoning. This chapter provides the logical foundation necessary for constructing and understanding proofs.

\subsection{Mathematical Statements and Basic Operators}

\begin{definition}[Mathematical Statement]
\label{def:mathematical_statement}
A \newterm{mathematical statement} (or logical statement) is a declarative sentence that is definitively either \textbf{True (T)} or \textbf{False (F)}. It is a fundamental axiom of logic that a statement cannot be both true and false simultaneously.
\end{definition}

\begin{example}[Examples of Logical Statements]
Consider the following declarations:
\begin{enumerate}[label=\textbf{\Alph*}:]
    \item \qt{For every real number $x$, $x^2 \ge 0$.}
    \item \qt{Every prime number $p \ge 3$ is odd.}
    \item \qt{Every odd number $p \ge 3$ is prime.}
\end{enumerate}
Statements A and B are True. Statement C is False (e.g., $9$ is odd but not prime).
\end{example}

\subsection{Negation, Conjunction, and Disjunction}

\begin{definition}[Negation]
\label{def:negation}
The \newterm{negation} of a statement $A$, denoted $\neg A$ (and sometimes $\overline{A}$), asserts that $A$ does not hold.
\[
\begin{array}{|c|c|}
\hline
A & \neg A \\
\hline
\text{T} & \text{F} \\
\text{F} & \text{T} \\
\hline
\end{array}
\]
\end{definition}

\begin{definition}[Conjunction / AND]
\label{def:conjunction}
\defterm{conjunction}
The statement $A \wedge B$ (read \qt{$A$ and $B$}, and sometimes written $A \,\&\, B$) is True if and only if both $A$ and $B$ are True.
\[
\begin{array}{|c|c|c|}
\hline
A & B & A \wedge B \\
\hline
\text{T} & \text{T} & \text{T} \\
\text{T} & \text{F} & \text{F} \\
\text{F} & \text{T} & \text{F} \\
\text{F} & \text{F} & \text{F} \\
\hline
\end{array}
\]
\end{definition}

\begin{example}[Conjunction Example]
For $x \in \mathbb{R}$, the statement $(x^2 = 1) \wedge (x \ge 0)$ is logically equivalent to $x = 1$.
\end{example}

\begin{definition}[Disjunction / OR]
\label{def:disjunction}
\defterm{disjunction}
The statement $A \vee B$ (read \qt{$A$ or $B$}) is True if \textit{at least one} of the statements $A$ or $B$ is True. This is an \textit{inclusive} OR.
\[
\begin{array}{|c|c|c|}
\hline
A & B & A \vee B \\
\hline
\text{T} & \text{T} & \text{T} \\
\text{T} & \text{F} & \text{T} \\
\text{F} & \text{T} & \text{T} \\
\text{F} & \text{F} & \text{F} \\
\hline
\end{array}
\]
\end{definition}

\begin{importantremark}
\label{rem:inclusive_or}
In everyday life, \qt{either $A$ or $B$} usually means one of the two but not both. In mathematics it means \qt{either $A$ or $B$, or both}: it could be only $A$, it could be only $B$, and it could be both $A$ and $B$.
\end{importantremark}

\label{ex:disjunction_interval}
The statement $(x > 1) \vee (x < 2)$ is True for all $x \in \mathbb{R}$, since every real number satisfies at least one of the two conditions. Conversely, $(x < 1) \vee (x > 2)$ is precisely the negation of the statement $1 \le x \le 2$, as \cref{fig:or_number_line} shows.
\end{example}

\subsection{Implications and Equivalences}

\begin{definition}[Implication]
\label{def:implication}
\defterm{implication}
The statement $A \implies B$ (\qt{$A$ implies $B$}) is defined to be False only when $A$ is True and $B$ is False. In all other cases, it is True.
\[
\begin{array}{|c|c|c|}
\hline
A & B & A \implies B \\
\hline
\text{T} & \text{T} & \text{T} \\
\text{T} & \text{F} & \text{F} \\
\text{F} & \text{T} & \text{T} \\
\text{F} & \text{F} & \text{T} \\
\hline
\end{array}
\]
\end{definition}

\begin{remark}
Note that $A \implies B$ is logically equivalent to $\neg A \vee B$. Consequently, if the premise $A$ is False, the implication is \textit{vacuously true} regardless of $B$. As Prof. Biran noted:
\[
(0=1) \implies \qt{\text{The Earth is flat}}
\]
is a mathematically \textbf{True} statement. Causality is irrelevant in formal logic.
\end{remark}

\begin{definition}[Equivalence]
\label{def:equivalence}
\defterm{equivalence of statements}
The statement $A \iff B$ (\qt{$A$ if and only if $B$}) holds when $A \implies B$ and $B \implies A$ are both True. This means $A$ and $B$ share the same truth value.
\[
\begin{array}{|c|c|c|}
\hline
A & B & A \iff B \\
\hline
\text{T} & \text{T} & \text{T} \\
\text{T} & \text{F} & \text{F} \\
\text{F} & \text{T} & \text{F} \\
\text{F} & \text{F} & \text{T} \\
\hline
\end{array}
\]
\end{definition}

\begin{example}[Equivalences]
\label{ex:equivalences}
Both of the following statements are True:
\[
(x^2 > 0) \iff (x \ne 0), \qquad (0 = 1) \iff \qt{\text{The Earth is flat}}.
\]
The second one is True because both sides are False. Again, causality is irrelevant. (In the first one, $x$ ranges over $\mathbb{R}$, as in the conjunction example above; the equivalence uses that a non-zero real has positive square, which is a fact about $\mathbb{R}$ and not a matter of logic).
\end{example}

\begin{proposition}[Contrapositive]
\label{prop:contrapositive}
Let $A, B$ be statements. Then
\[
(A \implies B) \iff (\neg B \implies \neg A).
\]
\end{proposition}

\label{exc:logic_contrapositive_reals}
Let $x, y \in \mathbb{R}$. Formulate the contrapositive of the implication
\[
x \ne y \implies \bigl( \exists \varepsilon > 0 : |x - y| > \varepsilon \bigr).
\]

\begin{exercise}[Negation and De Morgan's Laws for Statements and Sets]
\label{exer:de_morgan}
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item Let $A$ and $B$ be statements. Show the following logical equivalences:
    \begin{enumerate}[label=\textbf{(\roman*)}]
        \item $A$ is equivalent to $\neg(\neg A)$;
        \item $\neg(A \vee B)$ is equivalent to $\neg A \wedge \neg B$;
        \item $\neg(A \wedge B)$ is equivalent to $\neg A \vee \neg B$.
    \end{enumerate}
    \item Let $A$ and $B$ be two subsets contained in a universe $U$. Show the following set identities:
    \begin{enumerate}[label=\textbf{(\roman*)}]
        \item $(A \cup B)^c = A^c \cap B^c$;
        \item $(A \cap B)^c = A^c \cup B^c$.
    \end{enumerate}
\end{enumerate}
\exinfo{This exercise is Problem 2 of Problem Sheet 1, Linear Algebra I, Autumn Semester 2025.}
\exsol{sol:contrapositive}
\end{exercise}

\subsection{Predicates and Quantifiers}

A \newterm{predicate} $P(x)$ is a statement whose truth depends on a variable $x$. We use quantifiers to convert predicates into absolute statements.

\begin{definition}[Quantifiers]
\label{def:quantifiers}
\defterm{quantifier}
\begin{itemize}
    \item \textbf{Universal ($\forall$):} \qt{For every $x$ in $X$, $P(x)$ holds.}
    \item \textbf{Existential ($\exists$):} \qt{There exists at least one $x$ in $X$ such that $P(x)$ holds.}
\end{itemize}
\end{definition}

\begin{example}[Quantifiers over Integers]
\label{ex:quantifiers_integers}
Let $P(n)$ be the predicate $n = n^2$ over $\mathbb{Z}$.
\begin{itemize}
    \item $\forall n \in \mathbb{Z} : P(n)$ is False.
    \item $\exists n \in \mathbb{Z} : P(n)$ is True (take $n=1$).
    \item $\forall n \in \mathbb{Z} : \bigl( n = n^2 \implies (n = 0) \vee (n = 1) \bigr)$ is True.
\end{itemize}
\end{example}

\begin{exercise}[First-Order Logic and Negation of Quantifiers]
\label{exc:first_order_logic_naturals}
Let $\mathbb{N} = \{1, 2, 3, \dots\}$ denote the set of natural numbers.
\begin{enumerate}
    \item Formulate the statement \qt{there does not exist a largest natural number} in first-order logic.
    \item Formulate the statement \qt{for every natural number $n$, there exists a strictly larger natural number} in first-order logic.
    \item Show by formal transformation in first-order logic that both statements are logically equivalent.
\end{enumerate}
\exinfo{This exercise is Problem 1 of Problem Sheet 2, Linear Algebra I, Autumn Semester 2025.}
\exsol{sol:first_order_logic_naturals}
\end{exercise}

Quantifiers may of course be combined. For instance,
\[
\forall y \in \mathbb{R} : \bigl( y \ge 0 \implies (\exists x \in \mathbb{R} : x^2 = y) \bigr)
\]
is a True statement. In practice one writes this more compactly as
\begin{equation} \label{eq:sqrt_existence}
\forall \, 0 \le y \in \mathbb{R}, \ \exists x \in \mathbb{R} \ \text{ such that } \ x^2 = y.
\end{equation}

\begin{importantremark}[The Order of Quantifiers]
The order of quantifiers is non-commutative. Consider $X$ (ETH students), $Y$ (courses), and $A(x,y) := \qt{\text{student } x \text{ takes course } y}$.
\begin{itemize}
    \item $\forall x \in X, \exists y \in Y : A(x,y)$ means every student takes at least one course (True).
    \item $\exists y \in Y, \forall x \in X : A(x,y)$ means there is one specific course taken by \textit{every} student (False).
\end{itemize}
\end{importantremark}

\Cref{fig:quantifier_order} makes the asymmetry visible: reading the table row by row gives $\forall x \, \exists y$, whereas reading it column by column gives $\exists y \, \forall x$. Every row carries at least one mark, but no column is completely filled. The reader is encouraged to try the same interchange on \eqref{eq:sqrt_existence} and to decide what the swapped statement asserts there.

\subsection{Negation and Unique Existence}

\begin{proposition}[Negating Quantifiers]
\label{prop:negating_quantifiers}
\begin{align*}
\neg (\forall x \in X : A(x)) &\iff \exists x \in X : \neg A(x) \\
\neg (\exists x \in X : A(x)) &\iff \forall x \in X : \neg A(x)
\end{align*}
\end{proposition}

\begin{remark}[Shorthand Notation]
In practice, we often condense multiple quantifiers of the same type:
\begin{itemize}
    \item $\exists x \in X, \exists y \in X \implies \exists x,y \in X$
    \item $\forall x \in X, \forall y \in X \implies \forall x,y \in X$
\end{itemize}
\end{remark}

\begin{exercise}[Negation of Quantified Predicates with Relations]
\label{exc:logic_distribute_negation}
Let $x, y, z$ be variables and let $R, S$ be relations. Distribute the negation inside the parentheses in the following proposition:
\[
\neg \Bigl( \forall x \, \exists y \, \exists z : \bigl( R(x, y) \wedge R(z, x) \bigr) \vee \bigl( R(x, y) \wedge S(z, x) \bigr) \Bigr).
\]
\exinfo{This exercise is Problem 4 of Problem Sheet 1, Linear Algebra I, Autumn Semester 2025.}
\exsol{sol:logic_distribute_negation}
\end{exercise}

\begin{definition}[Unique Existence]
\label{def:unique_existence}
\defterm{unique existence}
The notation $\exists! x \in X : P(x)$ means there exists exactly one $x$ satisfying the predicate. Formally:
\[
(\exists x \in X : P(x)) \wedge (\forall x, y \in X : (P(x) \wedge P(y) \implies x = y))
\]
\end{definition}

\begin{example}[Unique Existence of Roots]
\label{ex:unique_existence_roots}
$\exists! x \in \mathbb{R} : x^2 = 4$ is False (since $x = \pm 2$). However, $\exists! x \in \mathbb{R} : (x \ge 0) \wedge (x^2 = 4)$ is True.
\end{example}

\subsection{Solutions to the Exercises}
\label{sec:solutions_logic}

\begin{exercisesolution}[Proof of \cref{prop:contrapositive} and of \cref{exer:de_morgan}]
\label{sol:contrapositive}
Each of the four claims is settled by writing out the truth table and checking that the two columns in question agree row by row. That is what an equivalence of statements amounts to, by \cref{def:equivalence}.

\textbf{The contrapositive.} Comparing the third column with the sixth:
\[
\begin{array}{|c|c|c|c|c|c|}
\hline
A & B & A \implies B & \neg B & \neg A & \neg B \implies \neg A \\
\hline
\text{T} & \text{T} & \text{T} & \text{F} & \text{F} & \text{T} \\
\text{T} & \text{F} & \text{F} & \text{T} & \text{F} & \text{F} \\
\text{F} & \text{T} & \text{T} & \text{F} & \text{T} & \text{T} \\
\text{F} & \text{F} & \text{T} & \text{T} & \text{T} & \text{T} \\
\hline
\end{array}
\]
The two columns are identical, which proves \cref{prop:contrapositive}. The single row where they could have differed is the second, and there both are False for the same reason: a true premise with a false conclusion.

\textbf{Double negation.} Here two rows suffice:
\[
\begin{array}{|c|c|c|}
\hline
A & \neg A & \neg(\neg A) \\
\hline
\text{T} & \text{F} & \text{T} \\
\text{F} & \text{T} & \text{F} \\
\hline
\end{array}
\]
The first and third columns agree, so $\neg(\neg A) \iff A$.

\textbf{The first De Morgan law.} Comparing the fourth column with the seventh:
\[
\begin{array}{|c|c|c|c|c|c|c|}
\hline
A & B & A \wedge B & \neg(A \wedge B) & \neg A & \neg B & \neg A \vee \neg B \\
\hline
\text{T} & \text{T} & \text{T} & \text{F} & \text{F} & \text{F} & \text{F} \\
\text{T} & \text{F} & \text{F} & \text{T} & \text{F} & \text{T} & \text{T} \\
\text{F} & \text{T} & \text{F} & \text{T} & \text{T} & \text{F} & \text{T} \\
\text{F} & \text{F} & \text{F} & \text{T} & \text{T} & \text{T} & \text{T} \\
\hline
\end{array}
\]

\textbf{The second De Morgan law.} Comparing the fourth column with the seventh:
\[
\begin{array}{|c|c|c|c|c|c|c|}
\hline
A & B & A \vee B & \neg(A \vee B) & \neg A & \neg B & \neg A \wedge \neg B \\
\hline
\text{T} & \text{T} & \text{T} & \text{F} & \text{F} & \text{F} & \text{F} \\
\text{T} & \text{F} & \text{T} & \text{F} & \text{F} & \text{T} & \text{F} \\
\text{F} & \text{T} & \text{T} & \text{F} & \text{T} & \text{F} & \text{F} \\
\text{F} & \text{F} & \text{F} & \text{T} & \text{T} & \text{T} & \text{T} \\
\hline
\end{array}
\]

\textbf{In words.} The first law says that for \qt{$A$ and $B$} to fail it is necessary and sufficient that \emph{at least one} of the two fails. The second says that for \qt{$A$ or $B$} to fail it is necessary and sufficient that \emph{both} fail, which is where the inclusive reading of \qt{or} recorded in \cref{rem:inclusive_or} makes itself felt: were \qt{or} exclusive, the second law would be false in the top row.

Two remarks worth carrying forward. First, the De Morgan laws are the statement-level version of \cref{prop:negating_quantifiers}: a universal quantifier is a long conjunction and an existential one a long disjunction, so negating either turns it into the other, exactly as here. Second, the contrapositive is the shape of every proof by contraposition in this book, of which the proof of \cref{prop:eigen_injective} in ch. 21 is a typical example.
\end{exercisesolution}

\begin{ainote}
The only chapter in the book with no linear algebra in it, and easy to read past.
Three of its results are used constantly afterwards. \Cref{prop:contrapositive}
is the shape of every proof that opens \qt{suppose the conclusion fails}.
\Cref{prop:negating_quantifiers} is what makes a negated definition usable, and
is applied silently whenever a family is shown \emph{not} to be independent by

The warning the chapter gives itself, in \cref{rem:inclusive_or}, is worth
repeating: \qt{or} is always inclusive. The second De Morgan law depends on it,
and so does the reading of \qt{$A$ or $B$} in every later statement.
\end{ainote}
